/**
 * 比對新舊物件陣列差異處
 *
 * ltdtOld與ltdtNew皆為必須參數，須為陣列，非陣列throw Error('invalid ltdtOld: must be an array')或Error('invalid ltdtNew: must be an array')，且各元素須皆為有效物件(即至少有一個鍵之非空物件)，有任一元素非有效物件(空物件{}、數字、字串、null、undefined、陣列、函數等，含陣列空洞)throw Error('invalid ltdtOld: every element must be a non-empty object')或Error('invalid ltdtNew: every element must be a non-empty object')，依ltdtOld(陣列、元素)、ltdtNew(陣列、元素)序，訊息不帶輸入值；本函數預期輸入物件陣列內各物件皆為有效物件，故不支援空物件列(空物件列無法判斷後續之有效物件由哪一筆而來)；空陣列為合法輸入，只有一邊為空者照常回傳差異結果，兩邊皆為空陣列時回傳{ diff: [], dfs: [] }(與正常輸出同形，無差異)；opt內無效的鍵一律改用預設
 * 欄位為兩側各列之自有可列舉字串鍵之聯集，取值只讀各列之自有同名欄位；缺該欄或其值為undefined時視為缺鍵，以opt.empty之標記表示；值等於缺鍵標記時無法與缺鍵區分，必要時以opt.empty自訂
 * 以儲存格文字比對、不比型別(1與'1'、true與'true'視為相同；需區分型別請用ltdtDiffByKey)，文字表示法同ltdtkeys2mat：字串與有限數字為原文，布林為'true'或'false'，null為'null'(與缺鍵區分)，NaN與±Infinity為字串，物件與陣列為鍵排序後之JSON(鍵之順序不同視為相同，dfs顯示排序後之JSON)，Date為ISO字串，函數視為null，BigInt為String(v)
 * 依列序對齊比對(以arrDiff之矩陣模式逐列再逐格比對，值內含任何字元皆不影響對齊；需依主鍵對齊請用ltdtDiffByKey)
 *
 * Unit Test: {@link https://github.com/yuda-lyu/wsemi/blob/master/test/ltdtDiff.test.mjs Github}
 * @memberOf wsemi
 * @param {Array} ltdtOld 輸入原始數據物件陣列，必須參數，非陣列或有元素非有效物件(含空物件{})throw Error，[]為合法輸入
 * @param {Array} ltdtNew 輸入更新數據物件陣列，必須參數，非陣列或有元素非有效物件(含空物件{})throw Error，[]為合法輸入
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {String} [opt.empty='∗empty∗'] 輸入缺鍵時之標記字串，非有效字串用預設，預設'∗empty∗'
 * @returns {Object} 回傳比對結果物件，回傳鍵值diff與dfs，其中diff為底層比對結果陣列(各段為{ count, added, removed, value }，value為該段各列之儲存格比對文字陣列，欄位依欄位聯集之順序)，dfs為各列比對結果陣列(每列為以欄位鍵為鍵、值為該儲存格比對結果之物件)；ltdtOld與ltdtNew皆為[]時回傳{ diff: [], dfs: [] }
 * @throws {Error} ltdtOld或ltdtNew非陣列時throw Error('invalid ltdtOld: must be an array')或Error('invalid ltdtNew: must be an array')；有任一元素非有效物件(含空物件{})時throw Error('invalid ltdtOld: every element must be a non-empty object')或Error('invalid ltdtNew: every element must be a non-empty object')；訊息皆不帶輸入值
 * @example
 *
 * let r
 * let ltdtOld
 * let ltdtNew
 *
 * if (true) {
 *     r = ltdtDiff([], [])
 *     console.log(r)
 *     // => { diff: [], dfs: [] }
 * }
 *
 * if (true) {
 *     try {
 *         ltdtDiff('x', [])
 *     }
 *     catch (err) {
 *         console.log(err.message)
 *     }
 *     // => invalid ltdtOld: must be an array
 *
 *     try {
 *         ltdtDiff([{ a: 1 }], null)
 *     }
 *     catch (err) {
 *         console.log(err.message)
 *     }
 *     // => invalid ltdtNew: must be an array
 *
 *     try {
 *         ltdtDiff([1], [2])
 *     }
 *     catch (err) {
 *         console.log(err.message)
 *     }
 *     // => invalid ltdtOld: every element must be a non-empty object
 *
 *     try {
 *         ltdtDiff([{ a: 1 }], [{ a: 1 }, null])
 *     }
 *     catch (err) {
 *         console.log(err.message)
 *     }
 *     // => invalid ltdtNew: every element must be a non-empty object
 *
 *     try {
 *         ltdtDiff([{ a: 1 }, {}], [{ a: 1 }]) //空物件列無效
 *     }
 *     catch (err) {
 *         console.log(err.message)
 *     }
 *     // => invalid ltdtOld: every element must be a non-empty object
 * }
 *
 * if (true) {
 *     ltdtOld = [
 *         { 'cn': '1', 'v1': 'a', 'v2': '0.506002098' },
 *         { 'cn': '2', 'v1': 'b', 'v2': '0.506002098' },
 *     ]
 *     ltdtNew = [
 *         { 'cn': 't1', 'v1': 'ta', 'v2': '0.506002098' },
 *         { 'cn': 't2', 'v1': 'tb', 'v2': '0.506002098' },
 *     ]
 *     r = ltdtDiff(ltdtOld, ltdtNew)
 *     console.dir(r.diff, { depth: null })
 *     // => [
 *     //   {
 *     //     count: 2,
 *     //     added: false,
 *     //     removed: true,
 *     //     value: [ [ '1', 'a', '0.506002098' ], [ '2', 'b', '0.506002098' ] ]
 *     //   },
 *     //   {
 *     //     count: 2,
 *     //     added: true,
 *     //     removed: false,
 *     //     value: [ [ 't1', 'ta', '0.506002098' ], [ 't2', 'tb', '0.506002098' ] ]
 *     //   }
 *     // ]
 *     for (let i = 0; i < r.dfs.length; i++) {
 *         let df = r.dfs[i]
 *         console.log(i, 'df', df)
 *     }
 *     // => 0 df {
 *     //   cn: { p: 'modify', k: 'cn', vo: '1', vn: 't1' },

 *     // }

 *     //     count: 1,
 *     //     added: true,
 *     //     removed: false,
 *     //     value: [ [ '1', '∗empty∗', '0.506002098', '' ] ]
 *     //   }
 *     // ]
 *     for (let i = 0; i < r.dfs.length; i++) {
 *         let df = r.dfs[i]
 *         console.log(i, 'df', df)
 *     }
 *     // => 0 df {
 *     //   cn: { p: '', k: 'cn', vo: '1', vn: '' },
 *     //   v1: { p: 'modify', k: 'v1', vo: '', vn: '∗empty∗' },
 *     //   v3: { p: '', k: 'v3', vo: '0.506002098', vn: '' },
 *     //   v2: { p: 'modify', k: 'v2', vo: '∗empty∗', vn: '' }
 *     // }
 * }
 *
 * if (true) {
 *     ltdtOld = [
 *         { 'hid': 'BH-01', 'sid': 'S-01', 'depth': '1-2', 'rsat': '18' },
 *         { 'hid': 'BH-01', 'sid': 'S-02', 'depth': '5-6', 'rsat': '17.5' },
 *         { 'hid': 'BH-01', 'sid': 'S-03', 'depth': '7-8', 'rsat': '17' },

 *     ltdtNew = [
 *         { 'hid': 'BH-01', 'sid': 'S-01', 'depth': '1-2', 'rsat': '18' },
 *         { 'hid': 'BH-01', 'sid': 'S-03', 'depth': '7-8', 'rsat': '17' },
 *         { 'hid': 'BH-01', 'sid': 'S-04', 'depth': '11-12', 'rsat': '19.5' },
 *     ]
 *     r = ltdtDiff(ltdtOld, ltdtNew)
 *     console.dir(r.diff, { depth: null })
 *     // => [
 *     //   {
 *     //     count: 1,
 *     //     added: false,
 *     //     removed: false,
 *     //     value: [ [ 'BH-01', 'S-01', '1-2', '18' ] ]
 *     //   },
 *     //   {
 *     //     count: 1,
 *     //     added: false,
 *     //     removed: true,
 *     //     value: [ [ 'BH-01', 'S-02', '5-6', '17.5' ] ]
 *     //   },
 *     //   {
 *     //     count: 1,
 *     //     added: false,
 *     //     removed: false,
 *     //     value: [ [ 'BH-01', 'S-03', '7-8', '17' ] ]
 *     //   },
 *     //   {
 *     //     count: 1,
 *     //     added: true,
 *     //     removed: false,
 *     //     value: [ [ 'BH-01', 'S-04', '11-12', '19.5' ] ]
 *     //   }
 *     // ]
 *     for (let i = 0; i < r.dfs.length; i++) {
 *         let df = r.dfs[i]
 *         console.log(i, 'df', df)

function ltdtDiff(ltdtOld, ltdtNew, opt = {}) {

    //check, ltdtOld與ltdtNew為必須參數(陣列且各元素為有效物件即非空物件, []合法), 無效throw(訊息不帶值), 依ltdtOld(陣列, 元素), ltdtNew(陣列, 元素)序
    //空物件列{}無效: 無法判斷後續之有效物件由哪一筆而來(使用者2026-11-12裁示; 2026-11-15第三批維持)
    //snap, 各列之快照(依索引走訪, 空洞視為undefined而無效; 每列只讀一次, 檢查之列即其後使用之列, 不呼叫輸入陣列之方法)
    let snap = (ltdt, name) => {
        let rows = []
        for (let i = 0; i < ltdt.length; i++) {
            let o = ltdt[i]
            if (!iseobj(o)) {
                throw new Error(`invalid ${name}: every element must be a non-empty object`)
            }
            rows.push(o)
        }
        return rows
    }
    if (!isarr(ltdtOld)) {
        throw new Error('invalid ltdtOld: must be an array')
    }
    ltdtOld = snap(ltdtOld, 'ltdtOld')
    if (!isarr(ltdtNew)) {
        throw new Error('invalid ltdtNew: must be an array')
    }
    ltdtNew = snap(ltdtNew, 'ltdtNew')
    if (size(ltdtOld) === 0 && size(ltdtNew) === 0) {
        return {
            diff: [],
            dfs: [],
        }
    }

    //empty
    let empty = get(opt, 'empty')
    if (!isestr(empty)) {
        empty = '∗empty∗'
    }

    //keysOld, keysNew
    let keysOld = getltdtkeys(ltdtOld)
    let keysNew = getltdtkeys(ltdtNew)

    //keys
    let keys = union(keysOld, keysNew)
    // console.log('keys', keys)

    //kp
    let kp = {}
    each(keys, (key, k) => {
        kp[k] = key
    })

    //matOld, matNew, 只讀自有可列舉同名欄位(同ltdtkeys2mat), 缺鍵或值為undefined給empty, 其餘原值交arrDiff依儲存格文字表示法比對(單一擁有者_cellText, 與ltdtkeys2mat同一規則; 物件與陣列為鍵排序後之JSON)
    let toRows = (ltdt) => {
        return ltdt.map((o) => {
            return keys.map((k) => {
                let v = _cellText.getOwn(o, k)
                return v === undefined ? empty : v
            })
        })
    }
    let matOld = toRows(ltdtOld)
    let matNew = toRows(ltdtNew)

    //arrDiff, 固定矩陣模式(逐列再逐格比對), 不轉傳其他opt(opt.dls、opt.dlq已移除)
    let r = arrDiff(matOld, matNew, { mode: 'matrix' })

    r.dfs = map(r.dfs, (r) => {

        //轉換指標k為欄位鍵key; 欄位鍵'__proto__'經_setOwnKey以自有欄位寫入(一般賦值會改結果物件之原型而使該欄消失)
        let rt = new Map()
        each(r, (c) => {
            c.k = kp[c.k]
            rt.set(c.k, c)
        })
        // console.log('rt', rt)

        //sort keys
        let ro = {}
        each(keys, (key) => {
            _setOwnKey(ro, key, rt.get(key))
        })
        // console.log('ro', ro)

        return ro
    })

    return r
}
