function mkErr(code, msg, extra = {}) {
    let err = new Error(maskUrl(msg)) //訊息內任何出現處之網址帳密皆遮蔽(含所併入之底層錯誤文字)
    err.code = code
    Object.assign(err, extra)
    _own.add(err)
    return err
}

//emit, 通知進度; 不帶接收者呼叫, 其拋錯與rejection皆忽略
function emit(ctx, ev) {
    let fn = ctx.onProgress
    if (fn === null) {
        return
    }
    try {
        let r = fn({ pkg: ctx.pkg, ...ev })
        if (r && isfun(r.catch)) {
            r.catch(() => {})
        }
    }
    catch (err) {}
}


//baseOfUrl, 網址路徑之最後一段(解碼), 作為壓縮檔或分割檔之檔名
function baseOfUrl(v) {
    try {
        let segs = new URL(v).pathname.split('/')
        return decodeURIComponent(segs[segs.length - 1])
    }
    catch (err) {
        return ''
    }
}


//bundleId, 檔案組之組識別(規格M6): 組內各檔依path排序(字串之碼位順序)後, 逐行'<sha256(小寫)>  <path>\n'(同sha256sum之輸出格式)串接之SHA-256
function bundleId(hash, files) {
    let ls = files.map((f) => {
        return { path: f.path, line: `${f.sha256}  ${f.path}\n` }
    })
    ls.sort((a, b) => {
        return a.path < b.path ? -1 : a.path > b.path ? 1 : 0
    })
    return hash(ls.map((v) => v.line).join(''))
}


//normManifest, 檢核清單(schema 1)並轉為內部結構; 每個欄位只讀取一次, 之後只用轉換後之值(檢查之值即使用之值); 不符拋E_MANIFEST
//nopt.hash為SHA-256函數(文字→64碼小寫十六進位, 算組識別), nopt.ref為無commit時自有檔之ref(預設主幹分支; 重取主幹清單時為所解析之提交)
function normManifest(raw, src, nopt) {
    let bad = (field, msg) => {
        return mkErr('E_MANIFEST', maskUrl(`fsEnsureBins: the manifest${src} is invalid: ${field} ${msg}`), { step: 'manifest', field: maskUrl(field) })
    }
    let keysOk = (o, keys, at) => {
        for (let k of Object.keys(o)) {
            if (!keys.includes(k)) {
                throw bad(`${at}.${k}`, `is not a field of schema 1 (allowed: ${keys.join(', ')})`)
            }
        }
    }
    let file = (o, at, keys) => {
        //壓縮檔、分割檔、解出檔共用之{ sha256, size }
        let sha256 = o.sha256
        let size = o.size
        if (!isStr(sha256) || !reHex64.test(sha256)) {
            throw bad(`${at}.sha256`, 'must be a string of 64 hexadecimal characters')
        }
        if (!isNonNegInt(size)) {
            throw bad(`${at}.size`, 'must be a non-negative integer')
        }
        return { sha256: sha256.toLowerCase(), size }
    }
    let source = (o, at) => {
        //自有檔(path)或廠商檔(url)之來源, 回傳{ path, url }(未給者為null)
        let p = o.path
        let u = o.url
        if ((p === undefined) === (u === undefined)) {
            throw bad(at, 'must have exactly one of path and url')
        }
        if (p !== undefined) {
            if (!isBinRel(p)) {
                throw bad(`${at}.path`, 'must be a relative path with / separators of letters, digits, spaces and ._+- (no empty, . or .. segments)')
            }
            return { path: p, url: null }
        }
        let e = isStr(u) ? binUrlProblem(u) : 'must be a string'
        if (e !== '') {
            throw bad(`${at}.url`, `${e}: ${isStr(u) ? u : typeof u}`)
        }
        return { path: null, url: new URL(u).href }
    }
    let partsOf = (parts, at) => {
        //分割檔: 各分割之來源與校驗值, 名稱(選填, 預設為path或url之最後一段)不分大小寫不得重複
        let r = []
        let pnames = new Set()
        let m = parts.length
        for (let j = 0; j < m; j++) {
            let p = parts[j]
            let atp = `${at}.parts[${j}]`
            if (!isObj(p)) {
                throw bad(atp, 'must be an object')
            }
            keysOk(p, ['name', 'path', 'url', 'sha256', 'size'], atp)
            let pname = p.name
            let s = source(p, atp)
            let fz = file(p, atp)
            if (pname === undefined) {
                pname = s.path !== null ? s.path.split('/').pop() : baseOfUrl(s.url)
            }
            if (!isBinName(pname)) {
                throw bad(`${atp}.name`, `must be a file name of letters, digits and ._+- (give name when the path or url does not end with one): ${String(pname)}`)
            }
            if (pnames.has(pname.toLowerCase())) {
                throw bad(`${atp}.name`, `duplicates another part name (case-insensitive): ${pname}`)
            }
            pnames.add(pname.toLowerCase())
            r.push({ name: pname, ...fz, ...s })
        }
        return r
    }
    let names = new Set()
    let nameOk = (name, at, what) => {
        //檔名與組名: 白名單、不分大小寫不得重複(兩者為結果物件之鍵)、不得為結果物件之不可列舉屬性名
        if (!isBinName(name)) {
            throw bad(at, 'must be a file name of letters, digits and ._+- (not a reserved device name)')
        }
        if (RESERVED.includes(name.toLowerCase())) {
            throw bad(at, `must not be ${RESERVED.join(', ')} (names of the properties of the result): ${name}`)
        }
        if (names.has(name.toLowerCase())) {
            throw bad(at, `duplicates another ${what} (case-insensitive, names of files and bundles share the keys of the result): ${name}`)
        }
        names.add(name.toLowerCase())
    }
    let noteOf = (v, at) => {
        if (v !== undefined && typeof v !== 'string') {
            throw bad(at, 'must be a string')
        }
        return v !== undefined ? v : null
    }
    let bundle = (b, at) => {
        //檔案組(規格M6): 組內各檔之path為組內相對路徑; 來源為repo檔(<dir>/<path>)、廠商網址(url)或壓縮檔之成員(archives[].extract.files)
        if (!isObj(b)) {
            throw bad(at, 'must be an object')
        }
        keysOk(b, ['name', 'dir', 'files', 'archives', 'note'], at)
        let name = b.name
        let dir = b.dir
        let bfiles = b.files
        let archs = b.archives
        let bnote = b.note
        nameOk(name, `${at}.name`, 'name')
        let note = noteOf(bnote, `${at}.note`)
        if (dir !== undefined && !isBinRel(dir)) {
            throw bad(`${at}.dir`, 'must be a relative path with / separators of letters, digits, spaces and ._+- (no empty, . or .. segments)')
        }
        if (bfiles !== undefined && !Array.isArray(bfiles)) {
            throw bad(`${at}.files`, 'must be an array')
        }
        if (archs !== undefined && !Array.isArray(archs)) {
            throw bad(`${at}.archives`, 'must be an array')
        }
        let nbf = bfiles !== undefined ? bfiles.length : 0
        let nba = archs !== undefined ? archs.length : 0
        if (nbf + nba === 0) {
            throw bad(at, 'must have a non-empty files or archives')
        }
        let all = [] //組內全部檔: { path, sha256, size, src(直接來源)或null, arch(壓縮檔之索引或-1) }
        let seen = new Map() //小寫path → path
        let addPath = (p, atp) => {
            if (!isBinRel(p)) {
                throw bad(`${atp}.path`, 'must be a relative path with / separators of letters, digits, spaces and ._+- (no empty, . or .. segments)')
            }
            let k = p.toLowerCase()
            if (seen.has(k)) {
                throw bad(`${atp}.path`, `duplicates another path in the bundle (case-insensitive): ${p}`)
            }
            seen.set(k, p)
        }
        for (let j = 0; j < nbf; j++) {
            let f = bfiles[j]
            let atf = `${at}.files[${j}]`
            if (!isObj(f)) {
                throw bad(atf, 'must be an object')
            }
            keysOk(f, ['path', 'url', 'sha256', 'size'], atf)
            let p = f.path
            let u = f.url
            addPath(p, atf)
            let fz = file(f, atf)
            let fname = p.split('/').pop()
            let s = null
            if (u !== undefined) {
                let e = isStr(u) ? binUrlProblem(u) : 'must be a string'
                if (e !== '') {
                    throw bad(`${atf}.url`, `${e}: ${isStr(u) ? u : typeof u}`)
                }
                s = { name: fname, ...fz, path: null, url: new URL(u).href }
            }
            else {
                if (dir === undefined) {
                    throw bad(atf, 'needs url, or dir of the bundle when the file is in the repository (its path in the repository is <dir>/<path>)')
                }
                let rp = `${dir}/${p}`
                if (!isBinRel(rp)) {
                    throw bad(`${atf}.path`, `makes a path in the repository that is too long: ${dir}/${p}`)
                }
                s = { name: fname, ...fz, path: rp, url: null }
            }
            all.push({ path: p, ...fz, src: s, arch: -1 })
        }
        let arcs = []
        for (let k = 0; k < nba; k++) {
            let a = archs[k]
            let ata = `${at}.archives[${k}]`
            if (!isObj(a)) {
                throw bad(ata, 'must be an object')
            }
            keysOk(a, ['path', 'url', 'parts', 'sha256', 'size', 'assembled', 'extract', 'note'], ata)
            let x = a.extract
            let parts = a.parts
            let assembled = a.assembled
            let anote = noteOf(a.note, `${ata}.note`)
            if (!isObj(x)) {
                throw bad(`${ata}.extract`, 'is required and must be an object')
            }
            keysOk(x, ['type', 'files'], `${ata}.extract`)
            let xtype = x.type
            let xfiles = x.files
            if (xtype !== 'zip') {
                throw bad(`${ata}.extract.type`, `must be 'zip'`)
            }
            if (!Array.isArray(xfiles) || xfiles.length === 0) {
                throw bad(`${ata}.extract.files`, 'must be a non-empty array')
            }
            let rec = { archive: null, parts: null, assembled: null, members: [], note: anote }
            if (parts !== undefined) {
                if (a.path !== undefined || a.url !== undefined || a.sha256 !== undefined || a.size !== undefined) {
                    throw bad(ata, 'with parts must not have path, url, sha256 or size (give them in each part and in assembled)')
                }
                if (!Array.isArray(parts) || parts.length === 0) {
                    throw bad(`${ata}.parts`, 'must be a non-empty array')
                }
                if (!isObj(assembled)) {
                    throw bad(`${ata}.assembled`, 'is required with parts and must be an object')
                }
                keysOk(assembled, ['type', 'sha256', 'size'], `${ata}.assembled`)
                if (assembled.type !== 'zip') {
                    throw bad(`${ata}.assembled.type`, `must be 'zip'`)
                }
                rec.assembled = file(assembled, `${ata}.assembled`)
                rec.parts = partsOf(parts, ata)
            }
            else {
                if (assembled !== undefined) {
                    throw bad(`${ata}.assembled`, 'is only for parts')
                }
                let s = source(a, ata)
                let fz = file(a, ata)
                let aname = s.path !== null ? s.path.split('/').pop() : baseOfUrl(s.url)
                if (!isBinName(aname)) {
                    throw bad(ata, `can not take a file name for the archive from its path or url: ${String(aname)}`)
                }
                rec.archive = { name: aname, ...fz, ...s }
            }
            let nm = xfiles.length
            for (let m = 0; m < nm; m++) {
                let mo = xfiles[m]
                let atm = `${ata}.extract.files[${m}]`
                if (!isObj(mo)) {
                    throw bad(atm, 'must be an object')
                }
                keysOk(mo, ['path', 'sha256', 'size'], atm)
                let p = mo.path
                addPath(p, atm)
                let fz = file(mo, atm)
                let fm = { path: p, ...fz, src: null, arch: k }
                all.push(fm)
                rec.members.push(fm)
            }
            arcs.push(rec)
        }
        //一檔之路徑不得為另一檔之上層資料夾(不分大小寫)
        for (let k of seen.keys()) {
            let segs = k.split('/')
            for (let i = 1; i < segs.length; i++) {
                let anc = segs.slice(0, i).join('/')
                if (seen.has(anc)) {
                    throw bad(at, `has a file whose path is a folder of another file: ${seen.get(anc)}`)
                }
            }
        }
        let id = bundleId(nopt.hash, all)
        return { kind: 'bundle', name, id, dir: dir !== undefined ? dir : null, note, files: all, archives: arcs }
    }
    try {
        if (!isObj(raw)) {
            throw bad('(root)', 'must be an object')
        }
        keysOk(raw, ['schema', 'pkg', 'mode', 'repo', 'commit', 'manifestPath', 'files', 'bundles', 'note', 'version'], '(root)')
        let schema = raw.schema
        let pkg = raw.pkg
        let mode = raw.mode
        let repo = raw.repo
        let commit = raw.commit
        let manifestPath = raw.manifestPath
        let files = raw.files
        let bundles = raw.bundles
        let note = raw.note
        let version = raw.version
        if (schema !== 1) {
            throw bad('schema', 'must be 1')
        }
        if (!isStr(pkg) || !rePkg.test(pkg)) {
            throw bad('pkg', 'must be a package name of lowercase letters, digits and ._-')
        }
        if (mode !== 'pinned' && mode !== 'floating') {
            throw bad('mode', `must be 'pinned' or 'floating'`)
        }
        if (!isStr(repo) || !reRepo.test(repo)) {
            throw bad('repo', `must be '<owner>/<repository>'`)
        }
        if (commit !== undefined && (!isStr(commit) || !reCommit.test(commit))) {
            throw bad('commit', 'must be 40 lowercase hexadecimal characters')
        }
        if (manifestPath !== undefined && !isBinRel(manifestPath)) {
            throw bad('manifestPath', 'must be a relative path with / separators')
        }
        if (note !== undefined && typeof note !== 'string') {
            throw bad('note', 'must be a string')
        }
        if (version !== undefined && typeof version !== 'string') {
            throw bad('version', 'must be a string')
        }
        if (files !== undefined && !Array.isArray(files)) {
            throw bad('files', 'must be an array')
        }
        if (bundles !== undefined && !Array.isArray(bundles)) {
            throw bad('bundles', 'must be an array')
        }
        let n = files !== undefined ? files.length : 0
        let nb = bundles !== undefined ? bundles.length : 0
        if (n + nb === 0) {
            throw bad('files', 'must be a non-empty array (or give a non-empty bundles)')
        }

        let entries = []
        for (let i = 0; i < n; i++) {
            let f = files[i]
            let at = `files[${i}]`
            if (!isObj(f)) {
                throw bad(at, 'must be an object')
            }
            keysOk(f, ['name', 'path', 'url', 'parts', 'sha256', 'size', 'assembled', 'extract', 'note'], at)
            let name = f.name
            let parts = f.parts
            let assembled = f.assembled
            let extract = f.extract
            let fnote = f.note
            nameOk(name, `${at}.name`, 'file name')
            fnote = noteOf(fnote, `${at}.note`)

            //extract
            let ext = null
            if (extract !== undefined) {
                if (!isObj(extract)) {
                    throw bad(`${at}.extract`, 'must be an object')
                }
                keysOk(extract, ['type', 'member', 'sha256', 'size'], `${at}.extract`)
                let type = extract.type
                let member = extract.member
                if (type !== 'zip') {
                    throw bad(`${at}.extract.type`, `must be 'zip'`)
                }
                if (!isBinRel(member)) {
                    throw bad(`${at}.extract.member`, 'must be a relative path with / separators of letters, digits, spaces and ._+- (no empty, . or .. segments)')
                }
                ext = { member, ...file(extract, `${at}.extract`) }
            }

            let e = { kind: 'file', name, final: null, direct: null, archive: null, parts: null, assembled: null, extract: ext, note: fnote }
            if (parts !== undefined) {
                //分割壓縮檔: 各分割與組合後之壓縮檔各有校驗值, 須解出
                if (f.path !== undefined || f.url !== undefined || f.sha256 !== undefined || f.size !== undefined) {
                    throw bad(at, 'with parts must not have path, url, sha256 or size (give them in each part and in assembled)')
                }
                if (!Array.isArray(parts) || parts.length === 0) {
                    throw bad(`${at}.parts`, 'must be a non-empty array')
                }
                if (ext === null) {
                    throw bad(`${at}.extract`, 'is required with parts')
                }
                if (!isObj(assembled)) {
                    throw bad(`${at}.assembled`, 'is required with parts and must be an object')
                }
                keysOk(assembled, ['type', 'sha256', 'size'], `${at}.assembled`)
                if (assembled.type !== 'zip') {
                    throw bad(`${at}.assembled.type`, `must be 'zip'`)
                }
                e.assembled = file(assembled, `${at}.assembled`)
                e.parts = partsOf(parts, at)
            }
            else {
                if (assembled !== undefined) {
                    throw bad(`${at}.assembled`, 'is only for parts')
                }
                let s = source(f, at)
                let fz = file(f, at)
                if (ext === null) {
                    e.direct = { name, ...fz, ...s }
                }
                else {
                    let aname = s.path !== null ? s.path.split('/').pop() : baseOfUrl(s.url)
                    if (!isBinName(aname)) {
                        throw bad(at, `can not take a file name for the archive from its path or url: ${String(aname)}`)
                    }
                    e.archive = { name: aname, ...fz, ...s }
                }
            }
            e.final = ext !== null ? { name, sha256: ext.sha256, size: ext.size } : { name, sha256: e.direct.sha256, size: e.direct.size }
            entries.push(e)
        }
        for (let i = 0; i < nb; i++) {
            entries.push(bundle(bundles[i], `bundles[${i}]`))
        }

        //自有檔(path, 檔案組為dir下之檔)之網址: 釘選模式須有commit, 主幹模式無commit時用主幹(或重取時所解析之提交)
        let srcsOf = (e) => {
            if (e.kind === 'bundle') {
                return [...e.files.map((f) => f.src), ...e.archives.flatMap((a) => [a.archive, ...(a.parts || [])])]
            }
            return [e.direct, e.archive, ...(e.parts || [])]
        }
        let ownUsed = entries.some((e) => {
            return srcsOf(e).some((o) => {
                return o !== null && o.path !== null
            })
        })
        if (mode === 'pinned' && ownUsed && commit === undefined) {
            throw bad('commit', 'is required in pinned mode when a file has a path')
        }
        let ref = commit !== undefined ? commit : isStr(get(nopt, 'ref')) ? nopt.ref : BRANCH
        for (let e of entries) {
            for (let o of srcsOf(e)) {
                if (o !== null && o.path !== null) {
                    o.url = `${RAW}/${repo}/${ref}/${encRel(o.path)}`
                }
            }
        }

        return {
            pkg,
            mode,
            repo,
            commit: commit !== undefined ? commit : null,
            manifestPath: manifestPath !== undefined ? manifestPath : 'bin/manifest.json',
            note: note !== undefined ? note : null,
            version: version !== undefined ? version : null,
            entries,
        }
    }
    catch (err) {
        if (isOwnErr(err)) {
            throw err
        }
        throw bad('(root)', `can not be read: ${errMsg(err)}`)
    }
}


//buildCtx, 檢核opt與環境變數; 所傳入之模組方法於此一次取出, 之後不再讀取opt
function buildCtx(opt) {
    let bad = (msg, cause) => {
        let extra = { step: 'argument' }
        if (cause !== undefined) {
            extra.cause = cause
        }
        return mkErr('E_ARG', maskUrl(`fsEnsureBins: ${msg}`), extra)
    }
    let o = {}
    o.refs = [] //所取出之原始方法(綁定前), 供共用判定: 同一模組物件於兩次呼叫之間被換了方法者不共用
    try {

        //fs
        o.fsRaw = get(opt, 'fs')
        o.fs = {}
        for (let k of fnsFs) {
            let f = get(o.fsRaw, k)
            if (!isfun(f)) {
                throw bad(`opt.fs is not an effective fs (needs ${fnsFs.join(', ')})`)
            }
            o.refs.push(f)
            o.fs[k] = f.bind(o.fsRaw)
        }

        //crypto, Readable, pipeline交fsDownloadFileVerifyCore; 本函數另用crypto.createHash、Readable.toWeb
        o.cryptoRaw = get(opt, 'crypto')
        let createHash = get(o.cryptoRaw, 'createHash')
        if (!isfun(createHash)) {
            throw bad(`opt.crypto is not an effective crypto (needs createHash)`)
        }
        o.createHash = createHash.bind(o.cryptoRaw)
        o.ReadableRaw = get(opt, 'Readable')
        let toWeb = get(o.ReadableRaw, 'toWeb')
        let fromWeb = get(o.ReadableRaw, 'fromWeb')
        let from = get(o.ReadableRaw, 'from')
        if (!isfun(toWeb) || !isfun(fromWeb) || !isfun(from)) {
            throw bad(`opt.Readable is not an effective stream.Readable (needs toWeb, fromWeb, from)`)
        }
        o.refs.push(createHash, toWeb, fromWeb, from)
        o.toWeb = toWeb.bind(o.ReadableRaw)
        o.pipeline = get(opt, 'pipeline')
        if (!isfun(o.pipeline)) {
            throw bad(`opt.pipeline is not a function`)
        }

        //spawn(解壓), path
        o.spawn = get(opt, 'spawn')
        if (!isfun(o.spawn)) {
            throw bad(`opt.spawn is not a function`)
        }
        let pathRaw = get(opt, 'path')
        let resolve = get(pathRaw, 'resolve')
        let join = get(pathRaw, 'join')
        if (!isfun(resolve) || !isfun(join)) {
            throw bad(`opt.path is not an effective path (needs resolve, join)`)
        }
        o.refs.push(resolve, join)
        o.path = { resolve: resolve.bind(pathRaw), join: join.bind(pathRaw) }

        //env, 環境變數統一於此讀取(未給為不設定)
        let env = get(opt, 'env')
        let envStr = (k) => {
            let v = get(env, k)
            return typeof v === 'string' ? v.trim() : ''
        }
        o.envCache = envStr('W_BIN_CACHE')
        o.envMirror = envStr('W_BIN_MIRROR')
        o.envOffline = envStr('W_BIN_OFFLINE')
        o.envSystemRoot = envStr('SystemRoot') || envStr('SYSTEMROOT') || envStr('windir')

        //選項, 給了就須為正確型別(undefined與null為未給)
        let given = (v) => {
            return v !== undefined && v !== null
        }
        o.pkgRoot = get(opt, 'pkgRoot')
        o.names = get(opt, 'names')
        o.cacheDir = get(opt, 'cacheDir')
        o.mirror = get(opt, 'mirror')
        o.offline = get(opt, 'offline')
        o.fetchRaw = get(opt, 'fetch')
        o.onProgress = get(opt, 'onProgress')
        o.floatingManifestUrl = get(opt, 'floatingManifestUrl')
        o.maxAttempts = get(opt, 'maxAttempts')
        o.retryDelayMs = get(opt, 'retryDelayMs')
        o.stallMs = get(opt, 'stallMs')
        o.totalMs = get(opt, 'totalMs')
        o.manifestMs = optNum(opt, 'stallMs', 30000, { min: 0, minOpen: true, timer: true }) //取主幹清單之上限同停滯毫秒數
        o.extractMs = optNum(opt, 'totalMs', 600000, { min: 0, minOpen: true, timer: true }) //解壓之上限同每檔整體時間上限
        if (!isStr(o.pkgRoot)) {
            throw bad(`opt.pkgRoot must be the root folder of the package (a non-empty string)`)
        }
        if (given(o.cacheDir) && typeof o.cacheDir !== 'string') {
            throw bad(`opt.cacheDir must be a string when given ('' is the default location)`)
        }
        if (given(o.mirror) && typeof o.mirror !== 'string') {
            throw bad(`opt.mirror must be a string when given`)
        }
        if (given(o.offline) && typeof o.offline !== 'boolean') {
            throw bad(`opt.offline must be a primitive boolean when given`)
        }
        if (given(o.fetchRaw) && !isfun(o.fetchRaw)) {
            throw bad(`opt.fetch is not a function`)
        }
        if (given(o.onProgress) && !isfun(o.onProgress)) {
            throw bad(`opt.onProgress is not a function`)
        }
        if (given(o.names)) {
            let l = typeof o.names === 'string' ? [o.names] : o.names
            if (!Array.isArray(l) || l.length === 0 || !l.every((v) => isStr(v))) {
                throw bad(`opt.names must be a non-empty string or an array of them when given`)
            }
            o.names = [...new Set(l)]
        }
        else {
            o.names = null
        }
        if (given(o.floatingManifestUrl)) {
            let u = null
            try {
                u = isStr(o.floatingManifestUrl) ? new URL(o.floatingManifestUrl) : null
            }
            catch (err) {}
            if (u === null || u.protocol !== 'https:' || u.username !== '' || u.password !== '') {
                throw bad(`opt.floatingManifestUrl must be an https url without credentials: ${isStr(o.floatingManifestUrl) ? o.floatingManifestUrl : typeof o.floatingManifestUrl}`)
            }
            o.floatingManifestUrl = u.href
        }
        else {
            o.floatingManifestUrl = null
        }
    }
    catch (err) {
        if (isOwnErr(err)) {
            throw err
        }
        throw bad(`opt can not be read: ${errMsg(err)}`, err)
    }

    //平台
    let pid = 0
    let win32 = false
    try {
        pid = get(globalThis, 'process.pid', 0) || 0
        win32 = get(globalThis, 'process.platform') === 'win32'
    }
    catch (err) {}

    //offline, 選項優先, 否則W_BIN_OFFLINE為1或true(不分大小寫)
    let offline = typeof o.offline === 'boolean' ? o.offline : /^(1|true)$/i.test(o.envOffline)

    //mirror, 選項優先(空字串為不使用), 否則W_BIN_MIRROR; 本地資料夾或http(s)基底網址
    let mirror = typeof o.mirror === 'string' ? o.mirror.trim() : o.envMirror
    let mirrorKind = ''
    let mirrorBase = ''
    if (mirror !== '') {
        if (/^[A-Za-z][A-Za-z0-9+.-]*:\/\//.test(mirror)) {
            let u = null
            try {
                u = new URL(mirror)
            }
            catch (err) {
                throw bad(`the mirror(W_BIN_MIRROR or opt.mirror) is not a valid url: ${mirror}`)
            }
            if (u.protocol !== 'http:' && u.protocol !== 'https:') {
                throw bad(`the mirror(W_BIN_MIRROR or opt.mirror) must be a local folder or an http(s) url: ${mirror}`)
            }
            if (u.search !== '' || u.hash !== '') {
                throw bad(`the mirror(W_BIN_MIRROR or opt.mirror) url must not have a query or hash: ${mirror}`)
            }
            if ((u.username !== '' || u.password !== '') && !isfun(o.fetchRaw)) {
                throw bad(`the mirror(W_BIN_MIRROR or opt.mirror) url must not contain credentials unless opt.fetch is given to handle them: ${mirror}`)
            }
            mirrorKind = 'http'
            mirrorBase = mirror.replace(/\/+$/, '')
        }
        else {
            mirrorKind = 'dir'
            mirrorBase = o.path.resolve(mirror)
        }
    }

    let fetchRaw = isfun(o.fetchRaw) ? o.fetchRaw : null
    return {
        fs: o.fs,
        fsRaw: o.fsRaw,
        cryptoRaw: o.cryptoRaw,
        createHash: o.createHash,
        ReadableRaw: o.ReadableRaw,
        toWeb: o.toWeb,
        pipeline: o.pipeline,
        spawn: o.spawn,
        path: o.path,
        fetchRaw,
        fetchFn: fetchRaw !== null
            ? fetchRaw
            : (u, init) => {
                return globalThis.fetch(u, init)
            },
        onProgress: isfun(o.onProgress) ? o.onProgress : null,
        pkgRoot: o.path.resolve(o.pkgRoot),
        names: o.names,
        cacheDir: isStr(o.cacheDir) ? o.path.resolve(o.cacheDir) : '',
        envCache: o.envCache !== '' ? o.path.resolve(o.envCache) : '',
        mirrorKind,
        mirrorBase,
        offline,
        floatingManifestUrl: o.floatingManifestUrl,
        tarProgs: win32 ? [o.path.join(o.envSystemRoot || 'C:\\Windows', 'System32', 'tar.exe')] : ['bsdtar', 'tar'],
        dl: { maxAttempts: o.maxAttempts, retryDelayMs: o.retryDelayMs, stallMs: o.stallMs, totalMs: o.totalMs },
        manifestMs: o.manifestMs,
        extractMs: o.extractMs,
        refs: o.refs,
        leftovers: [],
        fresh: [], //離線時記下「開始時尚不存在」之資料夾, 缺檔reject前把其中仍為空者清掉(見tidyFresh)
        trace: null, //觀察回調(opt.onTrace), 由入口取得後設定
        pid,
        win32,
        pkg: '',
        cacheRoot: '',
    }
}


//keyOf, 同程序內表之鍵: 絕對路徑(Windows不分大小寫)
function keyOf(ctx, fp) {
    return ctx.win32 ? fp.toLowerCase() : fp
}


//idOf, 檔案識別: 任一項改變即視為可能被改動, 須重新雜湊
function idOf(st) {
    return `${st.dev}:${st.ino}:${st.size}:${st.mtimeMs}:${st.ctimeMs}`
}


//lstatOrNull, 不存在者回傳null, 其他錯誤拋出
function lstatOrNull(ctx, fp) {
    try {
        return ctx.fs.lstatSync(fp)
    }
    catch (err) {
        let c = codeOf(err)
        if (c === 'ENOENT' || c === 'ENOTDIR') {
            return null
        }
        throw err
    }
}


//hashFile, 串流讀檔計算SHA-256與大小
async function hashFile(ctx, fp) {
    let h = ctx.createHash('sha256')
    let size = 0
    let rs = ctx.fs.createReadStream(fp)
    try {
        for await (let chunk of rs) {
            size += chunk.length
            h.update(chunk)
        }
    }
    finally {
        try {
            rs.destroy()
        }
        catch (err) {}
    }
    return { sha256: h.digest('hex'), size }
}


//isExpected, 檔案是否為期望內容(一般檔案且大小與SHA-256皆相符); 讀取失敗視為否
async function isExpected(ctx, fp, exp) {
    try {
        let st = ctx.fs.lstatSync(fp)
        if (!st.isFile() || st.size !== exp.size) {
            return false
        }
        let r = await hashFile(ctx, fp)
        return r.size === exp.size && r.sha256 === exp.sha256
    }
    catch (err) {
        return false
    }
}


//cacheErr, 快取之磁碟錯誤
function cacheErr(ctx, e, msg, cause) {
    let extra = { step: 'cache', pkg: ctx.pkg, bin: e.name }
    if (cause !== undefined) {
        extra.cause = cause
    }
    return mkErr('E_FS', `${labelOf(ctx, e)}: ${msg}`, extra)
}


//unlinkIf, 刪除不符清單之快取檔(或符號連結本身), 只在其識別與id相同時(他方剛換上之新檔不刪); 占用時重試
//回傳'deleted'(已刪或已不存在)或'changed'(識別已變, 未刪); 刪不掉拋E_FS(不符之檔不得留著而回報缺檔或成功)
async function unlinkIf(ctx, e, fp, id) {
    for (let i = 1; ; i++) {
        try {
            let st = lstatOrNull(ctx, fp)
            if (st === null) {
                return 'deleted'
            }
            if (idOf(st) !== id) {
                return 'changed'
            }
            ctx.fs.unlinkSync(fp)
            return 'deleted'
        }
        catch (err) {
            if (codeOf(err) === 'ENOENT') {
                return 'deleted'
            }
            if (!codesBusy.includes(codeOf(err)) || i >= 3) {
                throw cacheErr(ctx, e, `the cached file "${fp}" does not match the manifest and can not be removed: ${errMsg(err)}; remove it by hand`, err)
            }
            await sleep(100 * i)
        }
    }
}


//checkCache, 快取命中須校驗(D4): 一般檔案且大小與SHA-256皆符合本次清單; 本行程內已校驗且檔案識別未變、期望值相同者不重複雜湊
//雜湊期間檔案被換掉(識別改變)者重新檢查(至多CACHE_RECHECK次); 不符者刪除(識別未變才刪)並回傳false
async function checkCache(ctx, e, fp) {
    let key = keyOf(ctx, fp)
    let want = `${e.final.sha256}|${e.final.size}`
    for (let n = 0; n <= CACHE_RECHECK; n++) {
        let st = null
        try {
            st = lstatOrNull(ctx, fp)
        }
        catch (err) {
            throw cacheErr(ctx, e, `can not read the cached file "${fp}": ${errMsg(err)}`, err)
        }
        if (st === null) {
            _verified.delete(key)
            return false
        }
        if (!st.isSymbolicLink() && !st.isFile()) {
            throw cacheErr(ctx, e, `the cache path "${fp}" is not a file, remove it`)
        }
        let id = idOf(st)
        if (st.isFile() && st.size === e.final.size && _verified.get(key) === `${id}|${want}`) {
            return true
        }
        _verified.delete(key)
        let ok = false
        if (st.isFile() && st.size === e.final.size) {
            let r = null
            try {
                r = await hashFile(ctx, fp)
            }
            catch (err) {
                throw cacheErr(ctx, e, `can not read the cached file "${fp}": ${errMsg(err)}`, err)
            }
            ok = r.size === e.final.size && r.sha256 === e.final.sha256
            let st2 = null
            try {
                st2 = lstatOrNull(ctx, fp)
            }
            catch (err) {
                throw cacheErr(ctx, e, `can not read the cached file "${fp}": ${errMsg(err)}`, err) //不存在以外之讀取錯誤不當成換檔
            }
            if (st2 === null || !st2.isFile() || idOf(st2) !== id) {
                continue //雜湊期間被換掉或刪除: 所雜湊者不一定是現在之檔
            }
        }
        if (ok) {
            _verified.set(key, `${id}|${want}`)
            return true
        }
        let r = await unlinkIf(ctx, e, fp, id)
        emit(ctx, { step: 'cache', name: e.name, fp, hit: false, reason: st.isFile() ? 'mismatch' : 'symlink' })
        if (r === 'deleted') {
            return false
        }
    }
    //多次檢查期間仍持續變動: 無法穩定校驗, 不採用、不刪, 明確拒絕(不當成缺檔)
    throw cacheErr(ctx, e, `the cached file "${fp}" kept changing while being verified (${CACHE_RECHECK + 1} times), another process may be writing it; retry later`)
}


//verifyOpt, 交fsDownloadFileVerifyCore之選項
function verifyOpt(ctx, extra) {
    return {
        fs: ctx.fsRaw,
        crypto: ctx.cryptoRaw,
        Readable: ctx.ReadableRaw,
        pipeline: ctx.pipeline,
        maxAttempts: ctx.dl.maxAttempts,
        retryDelayMs: ctx.dl.retryDelayMs,
        stallMs: ctx.dl.stallMs,
        totalMs: ctx.dl.totalMs,
        useShare: false, //同程序內之共用由本函數以最終檔為鍵處理
        ...extra,
    }
}


//tagErr, 下載錯誤標註所屬之套件、檔名(檔案組為組名與組內路徑file)與來源(訊息前綴), 不改code
function tagErr(ctx, err, name, source, file = null) {
    if (!isOwnErr(err) && !isStr(codeOf(err))) {
        err = mkErr('E_FS', errMsg(err), { cause: err })
    }
    try {
        err.pkg = ctx.pkg
        err.bin = name
        err.source = source
        if (file !== null) {
            err.file = file
        }
        err.message = maskUrl(`fsEnsureBins[${ctx.pkg}] ${file !== null ? `bundle[${name}] file[${file}]` : `file[${name}]`} from ${source}: ${err.message}`)
    }
    catch (e) {}
    _tagged.add(err)
    return err
}


//addLeft, 下載核心回報之刪不掉之暫存檔(結果或錯誤之leftovers)併入本次之leftovers
function addLeft(ctx, e, x) {
    try {
        let l = get(x, 'leftovers')
        if (Array.isArray(l)) {
            for (let v of l) {
                ctx.left.push({ name: e.name, fp: get(v, 'fp'), error: maskUrl(get(v, 'error')) })
            }
        }
    }
    catch (err) {}
}


//mirrorPathOf, 鏡像內之位置: <鏡像>/<套件>/<sha256>/<name>; o.rel(<套件>之下之各段, 如檔案組之[<組識別>, ...path之各段])給了時為<鏡像>/<套件>/<rel>
function mirrorPathOf(ctx, o) {
    let segs = Array.isArray(o.rel) ? o.rel : [o.sha256, o.name]
    if (ctx.mirrorKind === 'dir') {
        return ctx.path.join(ctx.mirrorBase, ctx.pkg, ...segs)
    }
    if (ctx.mirrorKind === 'http') {
        return `${ctx.mirrorBase}/${encodeURIComponent(ctx.pkg)}/${segs.map((s) => encodeURIComponent(s)).join('/')}`
    }
    return ''
}


//localFetch, 自本地鏡像檔讀取之fetch(交fsDownloadFileVerifyCore, 使本地鏡像與網路下載同一套暫存、校驗、原子更名)
function localFetch(ctx, fp) {
    return async (u, init) => {
        let rs = ctx.fs.createReadStream(fp)
        let signal = get(init, 'signal')
        if (signal) {
            signal.addEventListener('abort', () => {
                rs.destroy()
            }, { once: true })
        }
        return new Response(ctx.toWeb(rs), { status: 200 })
    }
}


//fileOf, 事件與錯誤用之檔名: 檔案組之檔為其組內路徑(o.file), 否則為o.name
function fileOf(o) {
    return isStr(o.file) ? o.file : o.name
}


//fromMirror, 自鏡像取得o至dest並以清單校驗; 回傳true(取得)、null(鏡像無此檔)或拒用之說明(校驗不符、讀取或連線失敗)
async function fromMirror(ctx, e, o, dest) {
    let at = mirrorPathOf(ctx, o)
    let exp = { sha256: o.sha256, size: o.size }
    let fn = fileOf(o)
    try {
        if (ctx.mirrorKind === 'dir') {
            let st = lstatOrNull(ctx, at)
            if (st === null) {
                emit(ctx, { step: 'mirror', name: e.name, file: fn, from: at, result: 'miss' })
                if (ctx.trace) {
                    ctx.trace('mirror', { name: e.name, file: fn, result: 'miss' })
                }
                return null
            }
            if (!st.isFile()) {
                throw new Error(`"${at}" is not a file`)
            }
            let rv = await fsDownloadFileVerifyCore(at, dest, exp, verifyOpt(ctx, { fetch: localFetch(ctx, at), maxAttempts: 1, retryOnMismatch: false }))
            addLeft(ctx, e, rv)
        }
        else {
            //離線時不追隨重導(重導可指向鏡像以外之主機, 如GitHub), 重導即拒用; 傳入之fetch須遵守redirect選項
            let fn = ctx.fetchFn
            let f = ctx.offline
                ? (u, init) => {
                    return fn(u, { ...init, redirect: 'manual' })
                }
                : ctx.fetchRaw
            let rv = await fsDownloadFileVerifyCore(at, dest, exp, verifyOpt(ctx, { fetch: f, retryOnMismatch: false }))
            addLeft(ctx, e, rv)
        }
        emit(ctx, { step: 'mirror', name: e.name, file: fn, from: maskUrl(at), result: 'ok' })
        if (ctx.trace) {
            ctx.trace('mirror', { name: e.name, file: fn, result: 'ok' })
        }
        return true
    }
    catch (err) {
        addLeft(ctx, e, err) //拒用時下載核心刪不掉之暫存檔不得被吞掉
        //下載核心之E_FS、E_RENAME為寫入端(快取或暫存夾)之磁碟錯誤、E_ARG為參數錯誤, 皆不是鏡像之問題, 不續找下一來源(鏡像端之讀取或連線錯誤為E_NETWORK等)
        if (['E_FS', 'E_RENAME', 'E_ARG'].includes(codeOf(err))) {
            throw tagErr(ctx, err, e.name, 'mirror', e.kind === 'bundle' ? fn : null)
        }
        let status = get(err, 'status')
        if (codeOf(err) === 'E_HTTP' && (status === 404 || status === 410)) {
            emit(ctx, { step: 'mirror', name: e.name, file: fn, from: maskUrl(at), result: 'miss' })
            if (ctx.trace) {
                ctx.trace('mirror', { name: e.name, file: fn, result: 'miss' })
            }
            return null
        }
        let reason = maskUrl(`the mirror copy ${at} was rejected: ${isStr(codeOf(err)) ? `${codeOf(err)} ` : ''}${errMsg(err).split('\n')[0]}`)
        emit(ctx, { step: 'mirror', name: e.name, file: fn, from: maskUrl(at), result: 'rejected', error: reason })
        if (ctx.trace) {
            ctx.trace('mirror', { name: e.name, file: fn, result: 'rejected' })
        }
        return reason
    }
}


//missingOf, 離線缺檔之內部錯誤(由run彙整為E_OFFLINE_MISSING)
function missingOf(ctx, e, reasons) {
    let fp = ctx.path.join(ctx.cacheRoot, e.final.sha256, e.final.name)
    let alt = (o) => {
        return { name: o.name, sha256: o.sha256, size: o.size, url: o.url, mirrorPath: ctx.mirrorKind !== '' ? maskUrl(mirrorPathOf(ctx, o)) : null }
    }
    let src = e.direct !== null ? e.direct : null
    let m = {
        name: e.name,
        sha256: e.final.sha256,
        size: e.final.size,
        cachePath: fp,
        mirrorPath: ctx.mirrorKind !== '' ? maskUrl(mirrorPathOf(ctx, e.final)) : null,
        url: src !== null ? src.url : null,
        member: e.extract !== null ? e.extract.member : null,
        archive: e.archive !== null ? alt(e.archive) : null,
        parts: e.parts !== null ? e.parts.map(alt) : null,
        reasons,
        bundle: null,
    }
    if (ctx.trace) {
        ctx.trace('offline', { name: e.name })
    }
    return mkErr('E_OFFLINE_MISSING', `offline: ${e.name} is missing`, { step: 'offline', missing: [m] })
}


//missingBundleOf, 檔案組之離線缺檔(由run彙整為E_OFFLINE_MISSING): 列出全組(組須整組備妥), have為本次已取得(自鏡像)之組內路徑
function missingBundleOf(ctx, b, have, reasons) {
    let dirB = ctx.path.join(ctx.cacheRoot, b.id)
    let alt = (o) => {
        return { name: o.name, sha256: o.sha256, size: o.size, url: o.url, mirrorPath: ctx.mirrorKind !== '' ? maskUrl(mirrorPathOf(ctx, o)) : null }
    }
    let files = b.files.map((f) => {
        return {
            path: f.path,
            sha256: f.sha256,
            size: f.size,
            missing: !have.has(f.path),
            cachePath: ctx.path.join(dirB, ...f.path.split('/')),
            mirrorPath: ctx.mirrorKind !== '' ? maskUrl(mirrorPathOf(ctx, { rel: [b.id, ...f.path.split('/')] })) : null,
            url: f.src !== null ? f.src.url : null,
            archive: f.arch,
        }
    })
    let archives = b.archives.map((a) => {
        return { archive: a.archive !== null ? alt(a.archive) : null, parts: a.parts !== null ? a.parts.map(alt) : null, assembled: a.assembled, members: a.members.length }
    })
    let m = {
        name: b.name,
        sha256: null,
        size: b.files.reduce((t, f) => t + f.size, 0),
        cachePath: dirB,
        mirrorPath: ctx.mirrorKind !== '' ? maskUrl(mirrorPathOf(ctx, { rel: [b.id] })) : null,
        url: null,
        member: null,
        archive: null,
        parts: null,
        reasons,
        bundle: { id: b.id, files, archives },
    }
    if (ctx.trace) {
        ctx.trace('offline', { name: b.name })
    }
    return mkErr('E_OFFLINE_MISSING', `offline: ${b.name} is missing`, { step: 'offline', missing: [m] })
}


//obtain, 取得o至dest: 鏡像(有設定時)→GitHub或廠商網址(離線時不連); 任何來源皆以清單校驗; 回傳true(取得)
//離線而取不到時: 單一檔拋離線缺檔(missingOf), 檔案組之檔回傳false(由runBundle彙整全組)
async function obtain(ctx, e, o, dest, reasons) {
    let fn = fileOf(o)
    if (ctx.mirrorKind !== '') {
        let r = await fromMirror(ctx, e, o, dest)
        if (r === true) {
            return true
        }
        if (r !== null) {
            reasons.push(r)
        }
    }
    if (ctx.offline) {
        if (e.kind === 'bundle') {
            return false
        }
        throw missingOf(ctx, e, reasons)
    }
    emit(ctx, { step: 'download', name: e.name, file: fn, url: o.url })
    if (ctx.trace) {
        ctx.trace('download', { name: e.name, file: fn })
    }
    try {
        let rv = await fsDownloadFileVerifyCore(o.url, dest, { sha256: o.sha256, size: o.size }, verifyOpt(ctx, { fetch: ctx.fetchRaw }))
        addLeft(ctx, e, rv) //成功前之嘗試刪不掉之暫存檔
    }
    catch (err) {
        if (ctx.trace) {
            ctx.trace('downloaded', { name: e.name, file: fn, ok: false, code: codeStr(err) })
        }
        throw tagErr(ctx, err, e.name, o.url.startsWith(`${RAW}/`) ? 'github' : 'url', e.kind === 'bundle' ? fn : null)
    }
    if (ctx.trace) {
        ctx.trace('downloaded', { name: e.name, file: fn, ok: true, code: null })
    }
    return true
}


//writeFull, 同步寫完整個緩衝; 寫入量非正整數(零進度或異常回傳)即拋錯, 不無限迴圈
function writeFull(ctx, fd, buf) {
    let off = 0
    while (off < buf.length) {
        let n = ctx.fs.writeSync(fd, buf, off, buf.length - off)
        if (!Number.isSafeInteger(n) || n <= 0) {
            throw new Error(`writeSync returned ${String(n)} for ${buf.length - off} bytes`)
        }
        off += n
    }
}


//concatFiles, 依序串接分割檔為fpOut(排他建立)
async function concatFiles(ctx, fps, fpOut) {
    let fd = ctx.fs.openSync(fpOut, 'wx')
    try {
        for (let fp of fps) {
            let rs = ctx.fs.createReadStream(fp)
            try {
                for await (let chunk of rs) {
                    writeFull(ctx, fd, chunk)
                }
            }
            finally {
                rs.destroy()
            }
        }
    }
    finally {
        ctx.fs.closeSync(fd)
    }
}


//runTar, 執行解壓程式, 回傳{ code, stderr }、{ err, enoent }或{ timeout, exited }
//逾時(ctx.extractMs)則終止之, 並等其結束至多KILL_WAIT_MS(結束前不回傳, 免得清理暫存夾時解壓程式仍在寫入), exited為是否確認已結束
function runTar(ctx, name, prog, args, cwd) {
    return new Promise((resolve) => {
        let done = false
        let timedOut = false
        let timer = null
        let timerKill = null
        let stderr = ''
        let cp = null
        let fin = (r) => {
            if (done) {
                return
            }
            done = true
            clearTimeout(timer)
            clearTimeout(timerKill)
            resolve(r)
        }
        try {
            let spawn = ctx.spawn
            cp = spawn(prog, args, { cwd, windowsHide: true, stdio: ['ignore', 'ignore', 'pipe'] })
        }
        catch (err) {
            fin({ err, enoent: codeOf(err) === 'ENOENT' })
            return
        }
        cp.on('error', (err) => {
            if (timedOut) {
                return //終止失敗之error: 續等close或等待上限, 不提早回傳
            }
            fin({ err, enoent: codeOf(err) === 'ENOENT' })
        })
        if (cp.stderr) {
            cp.stderr.on('data', (d) => {
                if (stderr.length < 2000) {
                    stderr += String(d)
                }
            })
        }
        cp.on('close', (code) => {
            fin(timedOut ? { timeout: true, exited: true, stderr } : { code, stderr })
        })
        timer = setTimeout(() => {
            timedOut = true
            if (ctx.trace) {
                ctx.trace('kill', { name })
            }
            try {
                cp.kill()
            }
            catch (err) {}
            timerKill = setTimeout(() => {
                fin({ timeout: true, exited: false, stderr })
            }, KILL_WAIT_MS)
        }, ctx.extractMs)
    })
}


//labelOf, 錯誤訊息之前綴: 單一檔為file[<name>], 檔案組為bundle[<name>]
function labelOf(ctx, e) {
    return `fsEnsureBins[${ctx.pkg}] ${e.kind === 'bundle' ? 'bundle' : 'file'}[${e.name}]`
}


//extractErr, 解壓失敗(E_EXTRACT); what為所解之成員之說明, file為檔案組之組內路徑(單一成員時)
function extractErr(ctx, e, what, msg, cause, more = {}) {
    let extra = { step: 'extract', pkg: ctx.pkg, bin: e.name, ...more }
    if (cause !== undefined) {
        extra.cause = cause
    }
    return mkErr('E_EXTRACT', maskUrl(`${labelOf(ctx, e)}: can not extract ${what}: ${msg}`), extra)
}


//tarRun, 於cwd以tar(Windows內建tar.exe即bsdtar)自zipRel解出members至xRel; 參數皆為cwd內之相對路徑(不經命令列傳遞非ASCII之絕對路徑), 成員前加--
//成員多時分批執行(每批成員參數總長至多ARGS_CAP字元, 免逾命令列上限), 每批各通知extract與extracted; 失敗拋E_EXTRACT(逾時而解壓程式未確認結束者帶workBusy)
async function tarRun(ctx, e, cwd, zipRel, xRel, members, what) {
    ctx.fs.mkdirSync(ctx.path.join(cwd, xRel), { recursive: true })
    let batches = []
    let cur = []
    let len = 0
    for (let m of members) {
        if (cur.length > 0 && len + m.length + 1 > ARGS_CAP) {
            batches.push(cur)
            cur = []
            len = 0
        }
        cur.push(m)
        len += m.length + 1
    }
    batches.push(cur)
    for (let batch of batches) {
        //檔案組之錯誤帶file(組內路徑): 一批只有一個成員時為該成員, 多成員之批無法定位單一檔而不帶
        let more = e.kind === 'bundle' && batch.length === 1 ? { file: batch[0] } : {}
        emit(ctx, e.kind === 'bundle' ? { step: 'extract', name: e.name, members: batch.length } : { step: 'extract', name: e.name, member: batch[0] })
        if (ctx.trace) {
            ctx.trace('extract', { name: e.name })
        }
        let args = ['-x', '-f', zipRel, '-C', xRel, '--', ...batch]
        let r = null
        for (let prog of ctx.tarProgs) {
            r = await runTar(ctx, e.name, prog, args, cwd)
            if (!r.enoent) {
                break
            }
        }
        if (ctx.trace) {
            //result: 'error'為解壓程式不存在或無法啟動, 'timeout'為逾時(exited為終止後是否確認已結束), 'exit'為已結束(code為其離開碼)
            let result = 'enoent' in r ? 'error' : r.timeout ? 'timeout' : 'exit'
            ctx.trace('extracted', { name: e.name, result, code: result === 'exit' && typeof r.code === 'number' ? r.code : null, exited: result === 'timeout' ? r.exited === true : result === 'exit' ? true : null })
        }
        if (r.enoent) {
            throw extractErr(ctx, e, what, `the extraction program is not available (${ctx.tarProgs.join(' or ')}; Windows 10 1803 and later include tar.exe, other systems need bsdtar of libarchive)`, r.err, more)
        }
        if (r.err) {
            throw extractErr(ctx, e, what, errMsg(r.err), r.err, more)
        }
        if (r.timeout) {
            //終止後仍未確認結束: 保留暫存夾(解壓程式可能仍在寫入), 列入leftovers
            throw extractErr(ctx, e, what, `the extraction did not finish within ${ctx.extractMs}ms (opt.totalMs)${r.exited ? '' : `, and the extraction program did not exit within ${KILL_WAIT_MS}ms after being terminated, the work folder "${cwd}" is kept`}`, undefined, r.exited ? { ...more } : { ...more, workBusy: true })
        }
        if (r.code !== 0) {
            throw extractErr(ctx, e, what, `the extraction program exited with code ${r.code}: ${String(r.stderr).trim().slice(0, 500)}`, undefined, more)
        }
    }
}


//checkMember, 解出之檔須為一般檔案且大小與SHA-256符合清單; 不符拋E_EXTRACT(不在壓縮檔內、非一般檔案)或E_SIZE_MISMATCH、E_HASH_MISMATCH(step extract); rel為成員之路徑, file為檔案組之組內路徑
async function checkMember(ctx, e, fp, x, rel, file) {
    let more = file !== null ? { file } : {}
    let st = lstatOrNull(ctx, fp)
    if (st === null) {
        throw extractErr(ctx, e, `"${rel}"`, `the member is not in the archive`, undefined, more)
    }
    if (!st.isFile()) {
        throw extractErr(ctx, e, `"${rel}"`, `the member is not a regular file`, undefined, more)
    }
    let h = await hashFile(ctx, fp)
    if (h.size !== x.size) {
        throw mkErr('E_SIZE_MISMATCH', `${labelOf(ctx, e)}: the extracted "${rel}" has ${h.size} bytes, the manifest expects ${x.size} bytes`, { step: 'extract', pkg: ctx.pkg, bin: e.name, ...more, expected: { sha256: x.sha256, size: x.size }, actual: h })
    }
    if (h.sha256 !== x.sha256) {
        throw mkErr('E_HASH_MISMATCH', `${labelOf(ctx, e)}: the extracted "${rel}" has sha256 ${h.sha256}, the manifest expects ${x.sha256}`, { step: 'extract', pkg: ctx.pkg, bin: e.name, ...more, expected: { sha256: x.sha256, size: x.size }, actual: h })
    }
}


//extract, 於暫存夾內解出單一檔之成員(e.extract.member)並依清單extract校驗, 回傳解出之檔
async function extract(ctx, e, work, zipRel) {
    let x = e.extract
    await tarRun(ctx, e, work, zipRel, 'x', [x.member], `"${x.member}"`)
    let fp = ctx.path.join(work, 'x', ...x.member.split('/'))
    await checkMember(ctx, e, fp, x, x.member, null)
    return fp
}


//getArchive, 於cwd內取得壓縮檔(src.archive), 或各分割(src.parts, 各自校驗)依序組合並校驗src.assembled, 回傳zip於cwd內之相對路徑
//檔案組於離線取不到時回傳''(由runBundle彙整全組), 單一檔則由obtain拋離線缺檔
async function getArchive(ctx, e, src, cwd, reasons) {
    if (src.archive !== null) {
        let ok = await obtain(ctx, e, src.archive, ctx.path.join(cwd, 'archive', src.archive.name), reasons)
        return ok ? `archive/${src.archive.name}` : ''
    }
    let fps = []
    for (let p of src.parts) {
        let fp = ctx.path.join(cwd, 'parts', p.name)
        if (!await obtain(ctx, e, p, fp, reasons)) {
            return ''
        }
        fps.push(fp)
    }
    let zipRel = 'assembled.zip'
    let fpZip = ctx.path.join(cwd, zipRel)
    try {
        await concatFiles(ctx, fps, fpZip)
    }
    catch (err) {
        throw mkErr('E_FS', `${labelOf(ctx, e)}: can not assemble the parts: ${errMsg(err)}`, { step: 'assemble', pkg: ctx.pkg, bin: e.name, cause: err })
    }
    let h = await hashFile(ctx, fpZip)
    let a = src.assembled
    if (h.size !== a.size || h.sha256 !== a.sha256) {
        let code = h.size !== a.size ? 'E_SIZE_MISMATCH' : 'E_HASH_MISMATCH'
        throw mkErr(code, `${labelOf(ctx, e)}: the assembled archive has sha256 ${h.sha256} and ${h.size} bytes, the manifest expects ${a.sha256} and ${a.size} bytes`, { step: 'assemble', pkg: ctx.pkg, bin: e.name, expected: { sha256: a.sha256, size: a.size }, actual: h })
    }
    return zipRel
}


//place, 已校驗之解出檔原子移入快取; 失敗而最終檔已是期望內容(他方已放好)視為成功
async function place(ctx, e, fpX, dirFinal, fpFinal) {
    ctx.fs.mkdirSync(dirFinal, { recursive: true })
    for (let i = 1; i <= 5; i++) {
        try {
            ctx.fs.renameSync(fpX, fpFinal)
            return
        }
        catch (err) {
            if (await isExpected(ctx, fpFinal, e.final)) {
                return
            }
            if (!codesBusy.includes(codeOf(err)) || i === 5) {
                throw mkErr('E_RENAME', `${labelOf(ctx, e)}: can not move the extracted file to "${fpFinal}": ${errMsg(err)}`, { step: 'place', pkg: ctx.pkg, bin: e.name, cause: err })
            }
            await sleep(200 * i)
        }
    }
}


//mkWork, 於快取夾內建立本次專用之暫存夾(排他建立)
function mkWork(ctx, e) {
    let fail = (err) => {
        return mkErr('E_FS', `${labelOf(ctx, e)}: can not create a work folder in "${ctx.cacheRoot}": ${errMsg(err)}`, { step: 'prepare', pkg: ctx.pkg, bin: e.name, cause: err })
    }
    try {
        ctx.fs.mkdirSync(ctx.cacheRoot, { recursive: true })
    }
    catch (err) {
        throw fail(err)
    }
    for (let i = 0; i < 5; i++) {
        let fp = ctx.path.join(ctx.cacheRoot, `.work-${ctx.pid}-${rnd()}`)
        try {
            ctx.fs.mkdirSync(fp)
            return fp
        }
        catch (err) {
            if (codeOf(err) !== 'EEXIST' || i === 4) {
                throw fail(err)
            }
        }
    }
    return ''
}


//rmWork, 刪除本次之暫存夾(不跟隨連結); 刪不掉者記入本次取得之left(成功時為結果之leftovers、失敗時為錯誤之leftovers)並以進度通知, 不稱已清除
function rmWork(ctx, e, work) {
    try {
        ctx.fs.rmSync(work, { recursive: true, force: true, maxRetries: 3, retryDelay: 100 })
    }
    catch (err) {
        ctx.left.push({ name: e.name, fp: work, error: errMsg(err) })
        emit(ctx, { step: 'cleanup', name: e.name, fp: work, error: errMsg(err) })
        if (ctx.trace) {
            ctx.trace('cleanup', { name: e.name, ok: false })
        }
        return
    }
    //暫存夾已整個刪除: 先前回報之其內刪不掉之暫存(如下載核心之.part)已不存在, 自本次取得之leftovers移除
    let pre = `${work}${ctx.win32 ? '\\' : '/'}`
    let left = ctx.left
    for (let k = left.length - 1; k >= 0; k--) {
        let fp = get(left[k], 'fp')
        if (isStr(fp) && keyOf(ctx, fp).startsWith(keyOf(ctx, pre))) {
            left.splice(k, 1)
        }
    }
    if (ctx.trace) {
        ctx.trace('cleanup', { name: e.name, ok: true })
    }
}


//noteFresh, 離線時記下fd(本檔之<快取>/<sha256>夾)及其上層中「此刻尚不存在」之資料夾(自fd往上, 遇存在者即止)
//供缺檔reject前tidyFresh清掉其中仍為空者; 原本已存在之夾不記; 讀不到狀態(權限等)則不記
function noteFresh(ctx, fd) {
    let cur = fd
    for (;;) {
        let st = null
        try {
            st = lstatOrNull(ctx, cur)
        }
        catch (err) {
            return
        }
        if (st !== null) {
            return
        }
        if (!ctx.fresh.includes(cur)) {
            ctx.fresh.push(cur)
        }
        let up = ctx.path.join(cur, '..')
        if (up === cur) {
            return
        }
        cur = up
    }
}


//tidyFresh, 離線缺檔reject前, 把noteFresh所記之資料夾中仍為空者刪除(由深而淺, 用rmdirSync只刪空夾, 非空或刪不掉者保留不報錯); 回傳刪除之個數
//缺檔之嘗試(暫存夾、下載核心建立之<sha256>夾、鏡像失敗而留下之空夾)不得留下本次新建之空資料夾; 傳入之fs無rmdirSync者不處理
function tidyFresh(ctx) {
    let n = 0
    let fps = [...ctx.fresh].sort((a, b) => b.length - a.length)
    ctx.fresh.length = 0
    for (let fp of fps) {
        try {
            let rmdir = get(ctx.fsRaw, 'rmdirSync')
            if (isfun(rmdir)) {
                rmdir.call(ctx.fsRaw, fp)
                n++
            }
        }
        catch (err) {}
    }
    return n
}


//runEntry, 取得一個檔: 快取(校驗)→鏡像→GitHub或廠商網址; 壓縮檔於暫存夾內取得、組合、解出、校驗後才移入快取
async function runEntry(ctx, e, fd, fpFinal) {
    let hit = await checkCache(ctx, e, fpFinal)
    if (ctx.trace) {
        ctx.trace('cache', { name: e.name, hit })
    }
    if (hit) {
        emit(ctx, { step: 'cache', name: e.name, fp: fpFinal, hit: true })
        if (ctx.trace) {
            ctx.trace('ready', { name: e.name })
        }
        return fpFinal
    }
    if (ctx.offline) {
        noteFresh(ctx, fd) //離線: 記下本檔之<sha256>夾與快取根(及其上層)中尚不存在者, 缺檔reject前把仍為空者清掉
    }
    //取得後不記入行程內記憶(放入後至記錄前可能被換掉), 下次使用時再校驗一次
    let reasons = []
    if (e.extract === null) {
        await obtain(ctx, e, e.direct, fpFinal, reasons)
        emit(ctx, { step: 'ready', name: e.name, fp: fpFinal })
        if (ctx.trace) {
            ctx.trace('ready', { name: e.name })
        }
        return fpFinal
    }

    //鏡像可能已有解出之最終檔(由快取夾製成之鏡像)
    if (ctx.mirrorKind !== '') {
        let r = await fromMirror(ctx, e, e.final, fpFinal)
        if (r === true) {
            emit(ctx, { step: 'ready', name: e.name, fp: fpFinal })
            if (ctx.trace) {
                ctx.trace('ready', { name: e.name })
            }
            return fpFinal
        }
        if (r !== null) {
            reasons.push(r)
        }
    }

    //離線且無鏡像: 壓縮檔與分割檔皆無來源可取(離線不連網), 先於建暫存夾即回報缺檔, 不建立快取夾(唯讀之套件根亦得E_OFFLINE_MISSING而非建夾失敗)
    if (ctx.offline && ctx.mirrorKind === '') {
        throw missingOf(ctx, e, reasons)
    }

    let work = mkWork(ctx, e)
    if (ctx.trace) {
        ctx.trace('work', { name: e.name })
    }
    let keep = false
    try {
        let zipRel = await getArchive(ctx, e, e, work, reasons)
        let fpX = await extract(ctx, e, work, zipRel)
        await place(ctx, e, fpX, fd, fpFinal)
        if (ctx.trace) {
            ctx.trace('place', { name: e.name })
        }
    }
    catch (err) {
        keep = isOwnErr(err) && err.workBusy === true //解壓程式未確認結束: 不刪暫存夾
        throw err
    }
    finally {
        if (keep) {
            ctx.left.push({ name: e.name, fp: work, error: 'the extraction program did not exit, the work folder is kept' })
            if (ctx.trace) {
                ctx.trace('cleanup', { name: e.name, ok: false })
            }
        }
        else {
            rmWork(ctx, e, work)
        }
    }
    emit(ctx, { step: 'ready', name: e.name, fp: fpFinal })
    if (ctx.trace) {
        ctx.trace('ready', { name: e.name })
    }
    return fpFinal
}


//checkOne, 快取之檔案組內之一檔是否為期望內容(一般檔案且大小與SHA-256皆符合本次清單); 不刪除(不符之組由runBundle整組重取後替換)
//本行程內已校驗且檔案識別未變、期望值相同者不重複雜湊; 雜湊後再確認識別未變, 雜湊期間被換掉者重新檢查(至多CACHE_RECHECK次), 仍變動則E_FS
async function checkOne(ctx, b, f, fp) {
    let key = keyOf(ctx, fp)
    let want = `${f.sha256}|${f.size}`
    let rd = (err) => {
        return cacheErr(ctx, b, `can not read the cached file "${fp}": ${errMsg(err)}`, err)
    }
    for (let n = 0; n <= CACHE_RECHECK; n++) {
        let st = null
        try {
            st = lstatOrNull(ctx, fp)
        }
        catch (err) {
            throw rd(err)
        }
        if (st === null || !st.isFile() || st.size !== f.size) {
            _verified.delete(key)
            return false
        }
        let id = idOf(st)
        if (_verified.get(key) === `${id}|${want}`) {
            return true
        }
        _verified.delete(key)
        let r = null
        try {
            r = await hashFile(ctx, fp)
        }
        catch (err) {
            throw rd(err)
        }
        let st2 = null
        try {
            st2 = lstatOrNull(ctx, fp)
        }
        catch (err) {
            throw rd(err)
        }
        if (st2 === null || !st2.isFile() || idOf(st2) !== id) {
            continue //雜湊期間被換掉或刪除: 所雜湊者不一定是現在之檔
        }
        if (r.size === f.size && r.sha256 === f.sha256) {
            _verified.set(key, `${id}|${want}`)
            return true
        }
        return false
    }
    throw cacheErr(ctx, b, `the cached file "${fp}" kept changing while being verified (${CACHE_RECHECK + 1} times), another process may be writing it; retry later`)
}


//checkBundle, 快取之檔案組(規格M6、D4): <快取>/<組識別>須為資料夾(非連結), 組內每檔皆為一般檔案且大小與SHA-256符合本次清單(只校驗清單所列之檔)
//回傳'hit'(整組可用)、'miss'(組資料夾不存在)或'bad'(存在而不符: 缺檔、竄改、非一般檔案、非資料夾)
async function checkBundle(ctx, b, dirB) {
    let st = null
    try {
        st = lstatOrNull(ctx, dirB)
    }
    catch (err) {
        throw cacheErr(ctx, b, `can not read the cached bundle "${dirB}": ${errMsg(err)}`, err)
    }
    if (st === null) {
        return 'miss'
    }
    if (!st.isDirectory()) {
        return 'bad'
    }
    for (let f of b.files) {
        let ok = await checkOne(ctx, b, f, ctx.path.join(dirB, ...f.path.split('/')))
        if (!ok) {
            return 'bad'
        }
    }
    return 'hit'
}


//publishBundle, 整組原子發布: 已逐檔校驗之暫存組資料夾stage以一次更名成為dirB(<快取>/<組識別>)
//回傳'placed'(放入)、'replaced'(先移走不符之舊組後放入)或'exists'(目標已是整組校驗通過之組: 他方已發布, 本次之組捨棄)
//目標不符: 先將之移入本次暫存夾再更名; 所移走者於移走後再校驗, 若其實已是整組可用之組(他方於本次判定之後才放好)即移回並採用之;
//目標於移走前已消失(他方已移走)、或更名衝突(EEXIST、ENOTEMPTY)後目標已消失即重試更名; 最終失敗(含其後之讀取錯誤)時, 若曾移走舊組且目標仍不存在即盡力移回,
//移不回(目標已被他方放入或移回失敗)者隨暫存夾刪除(不符之組本應刪除, 規格D4; 所移走之好組移回失敗而本次之組已放入者為同內容之組); 移不走(使用中等)或更名失敗為E_RENAME
async function publishBundle(ctx, b, stage, dirB, work) {
    let aside = null //本次最先移走之舊組(於work內)
    let restore = () => {
        if (aside !== null) {
            try {
                if (lstatOrNull(ctx, dirB) === null) {
                    ctx.fs.renameSync(aside, dirB)
                }
            }
            catch (e) {}
        }
    }
    let fail = (msg, err) => {
        restore()
        return mkErr('E_RENAME', `${labelOf(ctx, b)}: ${msg}: ${errMsg(err)}`, { step: 'place', pkg: ctx.pkg, bin: b.name, cause: err })
    }
    let check = async (fp) => {
        //發布期間之校驗之讀取錯誤: 先移回舊組再拋出
        try {
            return await checkBundle(ctx, b, fp)
        }
        catch (err) {
            restore()
            throw err
        }
    }
    for (let i = 1; i <= 5; i++) {
        try {
            ctx.fs.renameSync(stage, dirB)
            return aside !== null ? 'replaced' : 'placed'
        }
        catch (err) {
            let s = await check(dirB)
            if (s === 'hit') {
                return 'exists'
            }
            if (s === 'bad') {
                let old = ctx.path.join(work, `old${i}`)
                try {
                    ctx.fs.renameSync(dirB, old)
                }
                catch (err2) {
                    if (codeOf(err2) === 'ENOENT') {
                        continue //判定之後已被他方移走: 重試更名
                    }
                    if ((await check(dirB)) === 'hit') {
                        return 'exists'
                    }
                    if (!codesBusy.includes(codeOf(err2)) || i === 5) {
                        throw fail(`the cached bundle "${dirB}" does not match the manifest and can not be replaced (close the programs using it, or remove it by hand)`, err2)
                    }
                    await sleep(200 * i)
                    continue
                }
                let first = aside === null
                if (first) {
                    aside = old //先登記, 其後之失敗皆可移回
                }
                //所移走者若已是整組可用之組(他方於本次判定之後才發布): 移回並採用, 不刪他方之組; 其校驗之讀取錯誤視為非好組(續以本次之組發布)
                let valid = false
                try {
                    valid = (await checkBundle(ctx, b, old)) === 'hit'
                }
                catch (err3) {}
                if (valid) {
                    try {
                        ctx.fs.renameSync(old, dirB)
                        if (first) {
                            aside = null
                        }
                        return 'exists'
                    }
                    catch (err4) {
                        if ((await check(dirB)) === 'hit') {
                            return 'exists'
                        }
                    }
                }
                continue
            }
            if (['EEXIST', 'ENOTEMPTY'].includes(codeOf(err))) {
                continue //更名衝突後目標已消失(他方已移走): 重試
            }
            if (!codesBusy.includes(codeOf(err)) || i === 5) {
                throw fail(`can not move the bundle to "${dirB}"`, err)
            }
            await sleep(200 * i)
        }
    }
    throw fail(`can not move the bundle to "${dirB}"`, new Error('retries exhausted'))
}


//runBundle, 取得一個檔案組(規格M6): ①快取(整組校驗) → ②鏡像之組資料夾<鏡像>/<套件>/<組識別>/<path>(逐檔) → ③其餘之直接來源檔(鏡像<sha256>/<檔名> → GitHub或廠商網址)
//與壓縮檔之成員(壓縮檔或各分割: 鏡像 → GitHub或廠商網址; 只解出尚缺之成員, 逐檔校驗); 皆於本次專用暫存夾內之組資料夾備妥且逐檔校驗後才整組原子發布(publishBundle)
//不符之舊組只在發布時被替換: 取得失敗時舊組原樣保留且不回傳; 離線而缺任何一檔時回報全組(missingBundleOf)
async function runBundle(ctx, b, dirB) {
    let state = await checkBundle(ctx, b, dirB)
    if (ctx.trace) {
        ctx.trace('cache', { name: b.name, hit: state === 'hit' })
    }
    if (state === 'hit') {
        emit(ctx, { step: 'cache', name: b.name, fp: dirB, hit: true })
        if (ctx.trace) {
            ctx.trace('ready', { name: b.name })
        }
        return dirB
    }
    if (state === 'bad') {
        emit(ctx, { step: 'cache', name: b.name, fp: dirB, hit: false, reason: 'mismatch' })
    }
    if (ctx.offline) {
        noteFresh(ctx, dirB) //離線: 記下尚不存在之組資料夾與快取根, 缺檔reject前把仍為空者清掉
    }
    let reasons = []
    let have = new Set()

    //離線且無鏡像: 無來源可取, 先於建暫存夾即回報缺檔(不建立快取夾)
    if (ctx.offline && ctx.mirrorKind === '') {
        throw missingBundleOf(ctx, b, have, reasons)
    }

    let work = mkWork(ctx, b)
    if (ctx.trace) {
        ctx.trace('work', { name: b.name })
    }
    let keep = false
    try {
        let stage = ctx.path.join(work, 'b')
        ctx.fs.mkdirSync(stage)
        let segs = (f) => {
            return f.path.split('/')
        }
        let dest = (f) => {
            return ctx.path.join(stage, ...segs(f))
        }

        //②鏡像之組資料夾(由快取製成之鏡像即為此結構); 本地鏡像無組資料夾時不逐檔查找
        if (ctx.mirrorKind !== '') {
            let absent = false
            if (ctx.mirrorKind === 'dir') {
                let at = mirrorPathOf(ctx, { rel: [b.id] })
                try {
                    absent = lstatOrNull(ctx, at) === null
                }
                catch (err) {}
                if (absent) {
                    emit(ctx, { step: 'mirror', name: b.name, file: null, from: at, result: 'miss' })
                    if (ctx.trace) {
                        ctx.trace('mirror', { name: b.name, file: null, result: 'miss' })
                    }
                }
            }
            if (!absent) {
                for (let f of b.files) {
                    let r = await fromMirror(ctx, b, { name: segs(f).pop(), file: f.path, sha256: f.sha256, size: f.size, rel: [b.id, ...segs(f)] }, dest(f))
                    if (r === true) {
                        have.add(f.path)
                    }
                    else if (r !== null) {
                        reasons.push(r)
                    }
                }
            }
        }

        //③直接來源之檔(repo檔、廠商網址)
        for (let f of b.files) {
            if (f.src === null || have.has(f.path)) {
                continue
            }
            if (await obtain(ctx, b, { ...f.src, file: f.path }, dest(f), reasons)) {
                have.add(f.path)
            }
        }

        //③壓縮檔之成員: 只解出尚缺者
        for (let k = 0; k < b.archives.length; k++) {
            let a = b.archives[k]
            let rest = a.members.filter((f) => {
                return !have.has(f.path)
            })
            if (rest.length === 0) {
                continue
            }
            let cwd = ctx.path.join(work, `a${k}`)
            ctx.fs.mkdirSync(cwd)
            let zipRel = await getArchive(ctx, b, a, cwd, reasons)
            if (zipRel === '') {
                continue //離線取不到
            }
            let what = rest.length === 1 ? `"${rest[0].path}"` : `${rest.length} members of ${a.archive !== null ? a.archive.name : `the archive of ${a.parts.length} parts`}`
            await tarRun(ctx, b, cwd, zipRel, 'x', rest.map((f) => f.path), what)
            for (let f of rest) {
                let fpX = ctx.path.join(cwd, 'x', ...segs(f))
                await checkMember(ctx, b, fpX, f, f.path, f.path)
                let d = dest(f)
                ctx.fs.mkdirSync(ctx.path.join(d, '..'), { recursive: true })
                ctx.fs.renameSync(fpX, d)
                have.add(f.path)
            }
        }

        if (have.size < b.files.length) {
            throw missingBundleOf(ctx, b, have, reasons) //只有離線會缺(連網時之失敗皆已拋出)
        }
        let result = await publishBundle(ctx, b, stage, dirB, work)
        emit(ctx, { step: 'publish', name: b.name, fp: dirB, result })
        if (ctx.trace) {
            ctx.trace('publish', { name: b.name, result })
        }
    }
    catch (err) {
        keep = isOwnErr(err) && err.workBusy === true //解壓程式未確認結束: 不刪暫存夾
        throw err
    }
    finally {
        if (keep) {
            ctx.left.push({ name: b.name, fp: work, error: 'the extraction program did not exit, the work folder is kept' })
            if (ctx.trace) {
                ctx.trace('cleanup', { name: b.name, ok: false })
            }
        }
        else {
            rmWork(ctx, b, work)
        }
    }
    emit(ctx, { step: 'ready', name: b.name, fp: dirB })
    if (ctx.trace) {
        ctx.trace('ready', { name: b.name })
    }
    return dirB
}


//normErr, 取得一個檔之失敗一律為帶code、step、pkg、bin之錯誤: 非本函數與下載核心之錯誤(如傳入之fs所拋)轉為E_FS
function normErr(ctx, e, err) {
    //下載核心之錯誤已由tagErr標註(以身分判定, 不以可被傳入元件偽造之pkg或code判定), 原樣回傳
    if (_tagged.has(err)) {
        return err
    }
    if (!isOwnErr(err)) {
        let c = codeOf(err)
        err = mkErr('E_FS', maskUrl(`${labelOf(ctx, e)}: ${isStr(c) ? `${c} ` : ''}${errMsg(err)}`), { step: 'prepare', cause: err })
    }
    if (err.code === 'E_OFFLINE_MISSING') {
        return err //離線缺檔由主流程彙整
    }
    try {
        if (err.pkg === undefined) {
            err.pkg = ctx.pkg
        }
        if (err.bin === undefined) {
            err.bin = e.name
        }
    }
    catch (e2) {}
    return err
}


//normLeft, 刪不掉之暫存之清單正規化: { name(缺者為name0), fp, error(遮蔽帳密) }, 依fp去重(Windows不分大小寫), 已不存在者不列
function normLeft(ctx, items, name0) {
    let r = []
    for (let v of items) {
        let fp = get(v, 'fp')
        if (!isStr(fp) || r.some((w) => keyOf(ctx, w.fp) === keyOf(ctx, fp))) {
            continue
        }
        let gone = false
        try {
            gone = lstatOrNull(ctx, fp) === null
        }
        catch (err) {}
        if (gone) {
            continue
        }
        let n = get(v, 'name')
        r.push({ name: isStr(n) ? n : name0, fp, error: maskUrl(get(v, 'error')) })
    }
    return r
}


//withLeftovers, 本次呼叫之最終錯誤附上本次各項(含主幹清單副本、重取前之嘗試)刪不掉之暫存; 有新增者另建錯誤物件(共用同一次取得者之錯誤物件為同一物件, 不改寫之)
function withLeftovers(ctx, err) {
    try {
        if (err === null || typeof err !== 'object' || (ctx.leftovers.length === 0 && !Array.isArray(err.leftovers))) {
            return err
        }
        let cur = Array.isArray(err.leftovers) ? err.leftovers : []
        let all = normLeft(ctx, [...cur, ...ctx.leftovers], '')
        //與現有之清單相同(同順序之同路徑)者不另建; 有新增或濾除(已不存在者)即另建
        let same = all.length === cur.length && all.every((v, k) => isStr(get(cur[k], 'fp')) && keyOf(ctx, cur[k].fp) === keyOf(ctx, v.fp))
        if (same) {
            return err
        }
        let e2 = new Error(err.message)
        for (let k of Object.getOwnPropertyNames(err)) {
            if (k !== 'message' && k !== 'stack') {
                e2[k] = err[k]
            }
        }
        e2.stack = err.stack
        e2.leftovers = all
        if (_own.has(err)) {
            _own.add(e2)
        }
        if (_tagged.has(err)) {
            _tagged.add(e2)
        }
        return e2
    }
    catch (e) {
        return err
    }
}


//ensureEntry, 同程序內同一最終檔(檔案組為同一組資料夾)、同樣來源設定之併發呼叫共用同一次取得; 回傳{ pm, left }, left為該次取得刪不掉之暫存(共用者共享)
function ensureEntry(ctx, e) {
    let isB = e.kind === 'bundle'
    let fd = ctx.path.join(ctx.cacheRoot, isB ? e.id : e.final.sha256)
    let fpFinal = isB ? fd : ctx.path.join(fd, e.final.name)
    let key = keyOf(ctx, fpFinal)
    let left = []
    let run = () => {
        let jctx = Object.create(ctx) //本次取得之脈絡: left為本次取得之leftovers
        jctx.left = left
        return (isB ? runBundle(jctx, e, fd) : runEntry(jctx, e, fd, fpFinal))
            .catch((err) => {
                err = normErr(ctx, e, err)
                if (err.code !== 'E_OFFLINE_MISSING') {
                    //下載核心之leftovers({ fp, error })與本次取得之left合併為{ name, fp, error }, 依fp去重, 已不存在者(隨暫存夾刪除)不列
                    try {
                        let all = [...(Array.isArray(err.leftovers) ? err.leftovers : []), ...left]
                        if (all.length > 0) {
                            err.leftovers = normLeft(ctx, all, e.name)
                        }
                    }
                    catch (e2) {}
                }
                if (ctx.trace && err.code !== 'E_OFFLINE_MISSING') {
                    ctx.trace('fail', { name: e.name, code: codeStr(err) })
                }
                throw err
            })
    }

    //共用條件: 影響來源與結果之值皆相同(清單項、套件、快取與鏡像位置、離線、下載與時限選項、解壓程式), 且所傳入之模組、fetch與其方法為同一物件
    let sig = null
    try {
        sig = JSON.stringify([e, ctx.pkg, ctx.cacheRoot, ctx.mirrorKind, ctx.mirrorBase, ctx.offline, ctx.dl, ctx.extractMs, ctx.tarProgs])
    }
    catch (err) {
        //下載選項含無法序列化之值(如BigInt): 不共用
        if (ctx.trace) {
            ctx.trace('job', { name: e.name, mode: 'solo' })
        }
        return { pm: run(), left }
    }
    let mods = [ctx.fsRaw, ctx.fetchRaw, ctx.spawn, ctx.cryptoRaw, ctx.ReadableRaw, ctx.pipeline, ...ctx.refs]
    let jobs = _jobs.get(key)
    if (jobs === undefined) {
        jobs = []
        _jobs.set(key, jobs)
    }
    let cur = jobs.find((job) => {
        return job.sig === sig && job.mods.length === mods.length && job.mods.every((m, i) => m === mods[i])
    })
    if (cur !== undefined) {
        if (ctx.trace) {
            ctx.trace('job', { name: e.name, mode: 'join' })
        }
        return { pm: cur.pm, left: cur.left }
    }
    if (ctx.trace) {
        ctx.trace('job', { name: e.name, mode: 'new' })
    }
    let job = { sig, mods, left, pm: null }
    job.pm = run()
        .finally(() => {
            let l = _jobs.get(key)
            if (l !== undefined) {
                let i = l.indexOf(job)
                if (i >= 0) {
                    l.splice(i, 1)
                }
                if (l.length === 0) {
                    _jobs.delete(key)
                }
            }
        })
    jobs.push(job)
    return { pm: job.pm, left }
}


//readCapped, 讀取回應本體為UTF-8文字(上限cap位元組)
async function readCapped(res, cap, hold) {
    let body = get(res, 'body')
    let chunks = []
    let n = 0
    let reader = body.getReader()
    hold.reader = reader //供逾時或失敗時取消(不依賴串流響應中止)
    for (;;) {
        let r = await reader.read()
        if (r.done) {
            break
        }
        n += r.value.length
        if (n > cap) {
            throw mkErr('E_MANIFEST', `the manifest is larger than ${cap} bytes`, { step: 'manifest' })
        }
        chunks.push(r.value)
    }
    let buf = new Uint8Array(n)
    let off = 0
    for (let c of chunks) {
        buf.set(c, off)
        off += c.length
    }
    hold.reader = null //已讀完, 不需取消
    return new TextDecoder('utf-8', { fatal: true }).decode(buf)
}


//cancelBody, 取消回應本體(或其reader)以釋放連線; 不等待其落定, 失敗不影響結果; why為通知觀察回調用之原因
function cancelBody(ctx, why, res, reader) {
    try {
        let p = reader ? reader.cancel() : res.body.cancel()
        if (p && isfun(p.catch)) {
            p.catch(() => {})
        }
    }
    catch (err) {}
    if (ctx.trace) {
        ctx.trace('cancel', { why })
    }
}


//fetchText, 取主幹清單(或鏡像之manifest-last.json、主幹之提交雜湊)之文字, 以manifestMs(同opt.stallMs)為整體上限, 本體上限cap位元組; init為另加之fetch選項(headers、redirect)
//逾時或失敗時取消未讀完之本體, 放棄等待後才落定之回應亦取消其本體
async function fetchText(ctx, url, init = {}, cap = MANIFEST_CAP) {
    let ac = new AbortController()
    let timer = null
    let done = false
    let hold = { reader: null }
    let pmTimeout = new Promise((resolve, reject) => {
        timer = setTimeout(() => {
            let e = mkErr('E_TIMEOUT', `getting the manifest did not finish within ${ctx.manifestMs}ms (opt.stallMs)`, { step: 'manifest' })
            ac.abort(e)
            reject(e)
        }, ctx.manifestMs)
    })
    pmTimeout.catch(() => {})
    let gotRes = null //fetch落定之回應
    let taken = false //回應已交本函數處理(其本體之取消由下方負責)
    try {
        let fn = ctx.fetchFn
        let pmFetch = Promise.resolve().then(() => {
            return fn(url, { ...init, signal: ac.signal })
        })
        pmFetch.then((r) => {
            gotRes = r
            if (done && !taken) {
                cancelBody(ctx, 'late', r, null) //放棄等待後才落定之回應(與finally之先後不定, 兩處擇一取消)
            }
        }, () => {})
        let res = await Promise.race([pmFetch, pmTimeout])
        taken = true
        if (!get(res, 'ok')) {
            cancelBody(ctx, 'status', res, null)
            throw mkErr('E_HTTP', `HTTP ${get(res, 'status')}`, { step: 'manifest', status: get(res, 'status') })
        }
        return await Promise.race([readCapped(res, cap, hold), pmTimeout])
    }
    catch (err) {
        if (isOwnErr(err)) {
            throw err
        }
        throw mkErr('E_NETWORK', maskUrl(errMsg(err)), { step: 'manifest', cause: err })
    }
    finally {
        done = true
        clearTimeout(timer)
        if (hold.reader !== null) {
            cancelBody(ctx, 'unread', null, hold.reader) //逾時、超過上限或讀取失敗時未讀完之本體
        }
        else if (gotRes !== null && !taken) {
            cancelBody(ctx, 'settled', gotRes, null) //逾時勝出而回應已落定(未交出)
        }
        ac.abort()
    }
}


//normOpt, normManifest之選項: hash(組識別用之SHA-256)與ref(無commit時自有檔之ref, null為主幹分支)
function normOpt(ctx, ref = null) {
    return {
        hash: (t) => {
            return ctx.createHash('sha256').update(t, 'utf8').digest('hex')
        },
        ref,
    }
}


//parseManifest, 文字解析並檢核, 須與基準清單同一套件與repo; ref為無commit時自有檔之ref(重取主幹清單時為所解析之提交)
function parseManifest(ctx, base, text, src, ref = null) {
    let raw = null
    try {
        raw = JSON.parse(String(text).replace(/^\uFEFF/, ''))
    }
    catch (err) {
        throw mkErr('E_MANIFEST', `fsEnsureBins: the manifest${src} is not valid JSON: ${errMsg(err)}`, { step: 'manifest' })
    }
    let m = normManifest(raw, src, normOpt(ctx, ref))
    if (m.pkg !== base.pkg || m.repo !== base.repo) {
        throw mkErr('E_MANIFEST', `fsEnsureBins: the manifest${src} is for ${m.pkg}(${m.repo}), not ${base.pkg}(${base.repo})`, { step: 'manifest', field: 'pkg' })
    }
    return m
}


//writeLast, 主幹清單之「最近一次成功」副本(紀錄, 供除錯與鏡像製作時知道檔案取自哪一版清單; 不作為校驗依據): 暫存檔排他建立、寫入後原子更名
//失敗(含寫入中途)時刪除本次建立之暫存檔, 以進度通知, 不影響結果
function writeLast(ctx, text) {
    let fp = ctx.path.join(ctx.cacheRoot, NAME_LAST)
    let tmp = `${fp}.${ctx.pid}.${rnd()}.part`
    let created = false
    let renamed = false
    try {
        ctx.fs.mkdirSync(ctx.cacheRoot, { recursive: true })
        let fd = ctx.fs.openSync(tmp, 'wx')
        created = true
        try {
            writeFull(ctx, fd, new TextEncoder().encode(text))
        }
        finally {
            ctx.fs.closeSync(fd)
        }
        ctx.fs.renameSync(tmp, fp)
        renamed = true
    }
    catch (err) {
        emit(ctx, { step: 'manifest', source: 'last', write: false, error: errMsg(err) })
    }
    finally {
        if (created && !renamed) {
            try {
                ctx.fs.unlinkSync(tmp)
            }
            catch (err) {
                ctx.leftovers.push({ name: NAME_LAST, fp: tmp, error: errMsg(err) }) //刪不掉者列入結果之leftovers, 不稱已清除
                emit(ctx, { step: 'cleanup', fp: tmp, error: errMsg(err) })
            }
        }
    }
    if (ctx.trace) {
        ctx.trace('last', { ok: renamed })
    }
}


//mirrorLast, 主幹模式之退路(規格M5、F5): 鏡像之<鏡像>/<套件>/manifest-last.json(由鏡像管理者寫入, 格式即主幹清單原文; 信任錨為鏡像管理者)
//同樣schema檢核且須同pkg與repo, 上限同主幹清單(1MB); 離線時http鏡像不追隨重導; 回傳採用之{ m, source: 'mirror-last', url }或null(無、拒用)
async function mirrorLast(ctx, base) {
    let at = mirrorPathOf(ctx, { rel: [NAME_LAST] })
    let note = (result, code, error) => {
        emit(ctx, { step: 'manifest', source: 'mirror-last', from: maskUrl(at), result, ...(error !== undefined ? { error } : {}) })
        if (ctx.trace) {
            ctx.trace('mirrorlast', { result, code })
        }
    }
    try {
        let text = ''
        if (ctx.mirrorKind === 'dir') {
            let st = lstatOrNull(ctx, at)
            if (st === null) {
                note('miss', null)
                return null
            }
            if (!st.isFile()) {
                throw mkErr('E_MANIFEST', `"${at}" is not a file`, { step: 'manifest' })
            }
            if (st.size > MANIFEST_CAP) {
                throw mkErr('E_MANIFEST', `the manifest is larger than ${MANIFEST_CAP} bytes`, { step: 'manifest' })
            }
            let buf = ctx.fs.readFileSync(at)
            if (buf.length > MANIFEST_CAP) {
                throw mkErr('E_MANIFEST', `the manifest is larger than ${MANIFEST_CAP} bytes`, { step: 'manifest' })
            }
            text = new TextDecoder('utf-8', { fatal: true }).decode(buf)
        }
        else {
            text = await fetchText(ctx, at, ctx.offline ? { redirect: 'manual' } : {})
        }
        let m = parseManifest(ctx, base, text, ` from the mirror ${maskUrl(at)}`)
        note('ok', null)
        return { m, source: 'mirror-last', url: maskUrl(at) }
    }
    catch (err) {
        let status = get(err, 'status')
        if (codeOf(err) === 'E_HTTP' && (status === 404 || status === 410)) {
            note('miss', null)
            return null
        }
        note('rejected', codeStr(err), maskUrl(`${codeOf(err) || ''} ${errMsg(err)}`.trim()))
        return null
    }
}


//trunkUrl, 主幹清單之網址: opt.floatingManifestUrl, 否則<RAW>/<repo>/master/<manifestPath>
function trunkUrl(ctx, base) {
    return ctx.floatingManifestUrl !== null ? ctx.floatingManifestUrl : `${RAW}/${base.repo}/${BRANCH}/${encRel(base.manifestPath)}`
}


//resolveManifest, 取本次採用之清單, 回傳{ m, source, url }: 釘選模式為傳入之清單(source 'base'); 主幹模式(規格M5、T8)依序為
//①主幹清單('remote', 離線時不取) → ②取不到(離線、連不到、無效)且有鏡像時鏡像之manifest-last.json('mirror-last') → ③傳入之基準清單('baseline')
//快取夾之「最近一次成功」副本存於快取夾, 其可信度同快取夾(能寫入者可同改副本與執行檔), 只作紀錄, 不作為退路, 以免使快取之校驗(D4)失效
async function resolveManifest(ctx, base) {
    if (base.mode !== 'floating') {
        return { m: base, source: 'base', url: null }
    }
    //信任錨為GitHub主幹(規格M5): opt.floatingManifestUrl只能是同一repo之GitHub原始檔網址(可改分支或路徑, 測試以opt.fetch改指本機)
    if (ctx.floatingManifestUrl !== null && !ctx.floatingManifestUrl.startsWith(`${RAW}/${base.repo}/`)) {
        throw mkErr('E_ARG', `fsEnsureBins[${base.pkg}]: opt.floatingManifestUrl must be under ${RAW}/${base.repo}/ (the trust anchor of the trunk manifest is GitHub), got ${ctx.floatingManifestUrl}`, { step: 'argument' })
    }
    if (!ctx.offline) {
        let url = trunkUrl(ctx, base)
        try {
            if (ctx.trace) {
                ctx.trace('trunk', { phase: 'fetch', code: null })
            }
            let text = await fetchText(ctx, url)
            let m = parseManifest(ctx, base, text, ` from ${url}`)
            writeLast(ctx, text)
            emit(ctx, { step: 'manifest', source: 'remote', url })
            if (ctx.trace) {
                ctx.trace('trunk', { phase: 'ok', code: null })
            }
            return { m, source: 'remote', url }
        }
        catch (err) {
            emit(ctx, { step: 'manifest', source: 'remote', url, error: maskUrl(`${codeOf(err) || ''} ${errMsg(err)}`.trim()) })
            if (ctx.trace) {
                ctx.trace('trunk', { phase: 'fail', code: codeStr(err) })
            }
        }
    }
    if (ctx.mirrorKind !== '') {
        let r = await mirrorLast(ctx, base)
        if (r !== null) {
            return r
        }
    }
    emit(ctx, { step: 'manifest', source: 'baseline' })
    return { m: base, source: 'baseline', url: null }
}


//refetchTrunk, 主幹模式下取得之檔與清單不符時重取一次主幹清單(規格M5, CDN約5分鐘不同步): raw.githubusercontent.com之快取鍵不含查詢字串且不理會no-cache(2026-10-29實測),
//故以GitHub API解析主幹之提交(<API>/repos/<repo>/commits/master, Accept: application/vnd.github.sha), 改取該提交之清單, 其無commit之自有檔亦取自該提交(同一版本之清單與檔);
//opt.floatingManifestUrl在<RAW>/<repo>/master/之下者同樣改取該提交之同一路徑, 其他分支或路徑者重取同一網址(不繞開CDN); 回傳{ m, source: 'remote', url }或null(失敗)
async function refetchTrunk(ctx, base) {
    let url = ''
    try {
        let ref = null
        //主幹分支之網址(預設, 或opt.floatingManifestUrl為<RAW>/<repo>/master/之下者; 分支段解碼後判定, 含%6Daster等編碼形式)改取所解析之提交; 其他分支或路徑重取同一網址
        let cur = trunkUrl(ctx, base)
        let segs = new URL(cur).pathname.split('/') //['', owner, repo, ref, ...path]
        let refSeg = ''
        try {
            refSeg = decodeURIComponent(segs[3] || '')
        }
        catch (err) {}
        if (refSeg !== BRANCH || segs.length < 5) {
            url = cur
        }
        else {
            let api = `${API}/repos/${base.repo}/commits/${BRANCH}`
            let sha = (await fetchText(ctx, api, { headers: { accept: 'application/vnd.github.sha' } }, COMMIT_CAP)).trim()
            if (!reCommit.test(sha)) {
                throw mkErr('E_MANIFEST', `${api} did not return a commit: ${sha.slice(0, 80)}`, { step: 'manifest' })
            }
            ref = sha
            url = `${RAW}/${base.repo}/${sha}/${segs.slice(4).join('/')}${new URL(cur).search}` //保留所給網址之查詢字串
        }
        let text = await fetchText(ctx, url)
        let m = parseManifest(ctx, base, text, ` from ${url}`, ref)
        writeLast(ctx, text)
        emit(ctx, { step: 'manifest', source: 'remote', url, refetch: true })
        if (ctx.trace) {
            ctx.trace('refetch', { phase: 'ok', code: null })
        }
        return { m, source: 'remote', url }
    }
    catch (err) {
        emit(ctx, { step: 'manifest', source: 'remote', url, refetch: true, error: maskUrl(`${codeOf(err) || ''} ${errMsg(err)}`.trim()) })
        if (ctx.trace) {
            ctx.trace('refetch', { phase: 'fail', code: codeStr(err) })
        }
        return null
    }
}


//offlineErr, 離線缺檔之固定格式錯誤: 缺哪個檔(名稱、sha256、大小)、應放到哪(快取與鏡像之完整路徑)、怎麼準備
function offlineErr(ctx, missing) {
    let ls = [`fsEnsureBins[${ctx.pkg}]: offline (W_BIN_OFFLINE or opt.offline), ${missing.length} external executable file(s) or bundle(s) are not in the cache or the mirror:`]
    for (let m of missing) {
        if (m.bundle !== null) {
            //檔案組: 須整組備妥, 列出全組各檔(缺者標missing)
            let b = m.bundle
            let nMiss = b.files.filter((f) => f.missing).length
            ls.push(`- ${m.name}: a bundle of ${b.files.length} files (${m.size} bytes, bundle id ${b.id}), ${nMiss} of them missing; the whole bundle is needed`)
            ls.push(`  put the whole bundle in: ${m.cachePath} (cache)`)
            ls.push(`  or in: ${m.mirrorPath !== null ? m.mirrorPath : `<W_BIN_MIRROR>/${ctx.pkg}/${b.id}`} (mirror${m.mirrorPath !== null ? '' : ', not set'})`)
            ls.push(`  files (path in the bundle: sha256, bytes):`)
            for (let f of b.files) {
                ls.push(`    ${f.path}: sha256 ${f.sha256}, ${f.size} bytes${f.missing ? ', missing' : ''}${f.url !== null ? `, source: ${f.url}` : ''}`)
            }
            for (let a of b.archives) {
                if (a.archive !== null) {
                    ls.push(`  ${a.members} of them are extracted from the archive ${a.archive.name}: sha256 ${a.archive.sha256}, ${a.archive.size} bytes, source: ${a.archive.url}; the archive may instead be put at: ${a.archive.mirrorPath !== null ? a.archive.mirrorPath : `<W_BIN_MIRROR>/${ctx.pkg}/${a.archive.sha256}/${a.archive.name}`} (mirror)`)
                }
                else {
                    ls.push(`  ${a.members} of them are extracted from an archive split into ${a.parts.length} parts, which may instead be put in the mirror:`)
                    for (let p of a.parts) {
                        ls.push(`    ${p.mirrorPath !== null ? p.mirrorPath : `<W_BIN_MIRROR>/${ctx.pkg}/${p.sha256}/${p.name}`}: sha256 ${p.sha256}, ${p.size} bytes, source: ${p.url}`)
                    }
                }
            }
            for (let r of m.reasons) {
                ls.push(`  note: ${r}`)
            }
            continue
        }
        ls.push(`- ${m.name}: sha256 ${m.sha256}, ${m.size} bytes`)
        ls.push(`  put it at: ${m.cachePath} (cache)`)
        if (m.mirrorPath !== null) {
            ls.push(`  or at: ${m.mirrorPath} (mirror)`)
        }
        else {
            ls.push(`  or at: <W_BIN_MIRROR>/${ctx.pkg}/${m.sha256}/${m.name} (mirror, not set)`)
        }
        if (m.url !== null) {
            ls.push(`  source: ${m.url}`)
        }
        if (m.archive !== null) {
            let a = m.archive
            ls.push(`  it is extracted from "${m.member}" of the archive ${a.name}: sha256 ${a.sha256}, ${a.size} bytes, source: ${a.url}; the archive may instead be put at: ${a.mirrorPath !== null ? a.mirrorPath : `<W_BIN_MIRROR>/${ctx.pkg}/${a.sha256}/${a.name}`} (mirror)`)
        }
        if (m.parts !== null) {
            ls.push(`  it is extracted from "${m.member}" of an archive split into ${m.parts.length} parts, which may instead be put in the mirror:`)
            for (let p of m.parts) {
                ls.push(`    ${p.mirrorPath !== null ? p.mirrorPath : `<W_BIN_MIRROR>/${ctx.pkg}/${p.sha256}/${p.name}`}: sha256 ${p.sha256}, ${p.size} bytes, source: ${p.url}`)
            }
        }
        for (let r of m.reasons) {
            ls.push(`  note: ${r}`)
        }
    }
    ls.push(`how to prepare: on a computer with internet access, set W_BIN_CACHE to an empty folder and call ensureBins() of this package (or use its functions that need these files), then bring that folder here and set W_BIN_MIRROR to it (the folder holds <package>/<sha256>/<name> and <package>/<bundle id>/<path>); or put each file above at the given path`)
    return mkErr('E_OFFLINE_MISSING', maskUrl(ls.join('\n')), { step: 'offline', pkg: ctx.pkg, missing })
}


/**
 * 後端nodejs依清單取得外部執行檔(快取、鏡像、GitHub)並以清單之SHA-256與大小校驗, 回傳各檔之絕對路徑, 可傳入fs、fetch、spawn與環境變數
 *
 * 規格: 外部執行檔之存放下載校驗與離線(制式規格, opensrcOper之dspec), 本函數為其§6之共用實作; 規則帳本R-BIN-ENSURE-01。各套件只提供清單、呼叫本函數取得路徑、匯出ensureBins(), 不得自寫下載、快取、鏡像、校驗。
 *
 * 清單manifest(schema 1)為物件或JSON檔路徑(相對路徑以opt.pkgRoot為基準): { schema: 1, pkg, mode: 'pinned'|'floating', repo: '<owner>/<repo>', commit(40碼小寫十六進位, 釘選模式有path之檔時必填), manifestPath(主幹模式之清單於repo內之路徑, 預設'bin/manifest.json'), files: [...], note, version }。files之每項: name(檔名, 英數與._+-, 不分大小寫不得重複), 來源為path(repo內之相對路徑, 自有檔, 網址為https://raw.githubusercontent.com/<repo>/<commit或master>/<path>)、url(廠商檔: 須為GitHub版本標籤之資產網址https://github.com/<owner>/<repo>/releases/download/<tag>/<檔>, 或GitHub原始檔網址且ref為40碼小寫提交[https://raw.githubusercontent.com/<owner>/<repo>/<commit>/<path>或https://github.com/<owner>/<repo>/raw/<commit>/<path>]; 須https, 不得帶帳密、查詢字串、片段、埠號, releases/latest、分支、標籤之原始檔網址與其他主機皆拒收, 規格M2、F15; parts、檔案組之url同)或parts(分割壓縮檔, 每項{ path或url, sha256, size, name(選填, 預設為path或url之最後一段) }, 須另給assembled: { type: 'zip', sha256, size }為組合後之校驗值), sha256(64碼十六進位)與size(位元組數)為該來源檔之校驗值(parts者不給), extract: { type: 'zip', member(壓縮檔內之相對路徑), sha256, size }為解出之檔之校驗值, note(選填)。不認得之欄位、型別不符、名稱或路徑不合白名單皆為E_MANIFEST。檔案組(規格M6; 頂層bundles, 有bundles時files可省略): [{ name(組名), dir(選填, 組內repo檔所在之repo資料夾), files: [{ path(組內相對路徑), url(選填, 廠商網址; 未給者為repo檔<dir>/<path>), sha256, size }], archives: [{ path、url或parts＋assembled(壓縮檔之來源, 同files之壓縮檔), sha256, size, extract: { type: 'zip', files: [{ path(壓縮檔內之路徑, 即組內路徑), sha256, size }] }, note }], note }], files與archives至少一個非空; 組內路徑不分大小寫不得重複, 一檔之路徑不得為另一檔之上層資料夾。files之name與檔案組之name共用結果物件之鍵, 不分大小寫不得重複, 且不得為leftovers、manifest、entries、bundles(不分大小寫, 結果物件之不可列舉屬性名)。
 *
 * 取得順序: ①快取: 位置為<快取>/<sha256>/<name>(內容定址, sha256為最終檔之值, 壓縮檔者為extract之值), 快取資料夾為opt.cacheDir, 否則環境變數W_BIN_CACHE時為<W_BIN_CACHE>/<pkg>, 否則<opt.pkgRoot>/bin-cache; 命中亦校驗大小與SHA-256, 本行程內已校驗且檔案識別(裝置、inode、大小、修改與變更時間)未變者不重複雜湊, 不符者刪除(識別未變才刪)後重新取得。②鏡像: opt.mirror, 否則環境變數W_BIN_MIRROR, 為本地資料夾或http(s)基底網址, 結構為<鏡像>/<pkg>/<sha256>/<name>; 壓縮檔者先找解出之最終檔, 再找壓縮檔(<name>為其path或url之最後一段)或各分割檔; 鏡像之檔不符清單即拒用(不留檔)並續找下一來源。③GitHub或廠商網址(離線時不連)。任何來源皆經fsDownloadFileVerify之暫存檔、校驗、原子更名寫入, 不讀到半檔。檔案組: 位置為<快取>/<組識別>/<path>, 組識別為組內各檔依path排序(字串之碼位順序)後逐行'<sha256>  <path>\n'(同sha256sum之輸出)串接之SHA-256(64碼小寫); 快取命中須組資料夾(非連結)內清單所列之每檔皆為一般檔案且大小與SHA-256符合(只校驗清單所列之檔), 任一檔不符(竄改、缺、非一般檔案)即整組重取: 依序為鏡像之組資料夾<鏡像>/<pkg>/<組識別>/<path>(逐檔; 本地鏡像無組資料夾時略過)、各直接來源檔(鏡像<sha256>/<path之最後一段> → GitHub或廠商網址)、壓縮檔之成員(壓縮檔或各分割: 鏡像 → GitHub或廠商網址, 只解出尚缺之成員, 成員多時分批執行tar, 逐檔校驗), 全部於本次暫存夾內備妥並逐檔校驗後才以一次更名整組原子發布; 不符之舊組於發布時才被替換(移入暫存夾後隨之刪除, 使用中而移不走為E_RENAME), 取得失敗時舊組原樣保留且不回傳; 他方已先發布且整組校驗通過者採用之。
 *
 * 離線: opt.offline, 否則環境變數W_BIN_OFFLINE為1或true時, 只用快取與鏡像, 缺檔時reject E_OFFLINE_MISSING, 訊息列出每個缺檔之名稱、sha256、大小、應放之快取與鏡像完整路徑、來源網址與準備方法, 錯誤物件之missing為[{ name, sha256, size, cachePath, mirrorPath, url, member, archive, parts, reasons, bundle }](bundle於單一檔為null)。檔案組缺任何一檔時列出全組(須整組備妥): 其項之sha256為null、size為全組總大小、cachePath與mirrorPath為組資料夾, bundle為{ id, files: [{ path, sha256, size, missing(本次未取得), cachePath, mirrorPath, url, archive(壓縮檔之索引或-1) }], archives: [{ archive, parts, assembled, members }] }。
 *
 * 主幹模式(mode為floating): 先取主幹清單(opt.floatingManifestUrl, 否則https://raw.githubusercontent.com/<repo>/master/<manifestPath>, 上限1MB, 時限同opt.stallMs), 同樣檢核schema且須為同一pkg與repo; 成功時其副本存於<快取>/manifest-last.json(紀錄, 不作為校驗依據); 失敗(含離線、無效)時, 有鏡像者先用鏡像之<鏡像>/<pkg>/manifest-last.json(鏡像管理者寫入之主幹清單原文, 信任錨為鏡像管理者; 同樣檢核, 上限1MB, 離線時http鏡像不追隨重導; 規格M5、F5), 無或拒用時用傳入之基準清單; 快取夾之副本只作紀錄, 不作退路(F6)。取得之檔(含解出、組合之檔)與清單不符(E_HASH_MISMATCH、E_SIZE_MISMATCH)、非離線且本次採用之清單亦為主幹模式時重取一次主幹清單再試(規格M5、F7, CDN約5分鐘不同步): raw.githubusercontent.com之CDN快取鍵不含查詢字串且不理會no-cache(2026-10-29實測), 故以https://api.github.com/repos/<repo>/commits/master(Accept: application/vnd.github.sha)解析主幹之提交, 改取該提交之清單, 其無commit之自有檔亦取自該提交; opt.floatingManifestUrl在https://raw.githubusercontent.com/<repo>/master/之下者同樣改取該提交之同一路徑, 其他分支或路徑者重取同一網址(不繞開CDN); 重取失敗時以原錯誤失敗, 重取後仍不符則以其錯誤失敗。主幹清單之信任錨為GitHub主幹(規格M5), 故opt.floatingManifestUrl只能是https://raw.githubusercontent.com/<repo>/之下(同一repo, 可改分支或路徑, 否則E_ARG), 套件之ensureBins不應轉傳使用者之值。
 *
 * 壓縮檔: 於快取夾內本次專用之暫存夾(.work-<pid>-<亂數>)取得壓縮檔(或各分割檔各自校驗後依序組合並校驗assembled), 以tar解出成員(Windows為<SystemRoot>\System32\tar.exe即bsdtar, 其他系統為bsdtar或tar, 須支援zip, 參數皆為暫存夾內之相對路徑, 成員前加--), 解出之檔須為一般檔案且符合extract之大小與SHA-256, 才原子移入快取; 暫存夾於結束時刪除, 刪不掉者(含下載核心刪不掉之暫存檔、主幹清單副本之暫存檔)列於結果之leftovers(不可列舉之屬性; 失敗時為錯誤物件之leftovers; 共用同一次取得者皆得到)。解壓時限同opt.totalMs, 逾時終止後等其結束至多5秒, 仍未結束則保留暫存夾並列入leftovers。解壓失敗為E_EXTRACT。
 *
 * 清單與選項之字串一律須為原始字串(包裝物件new String與new Boolean不接受), 檢查之值即使用之值。離線時http(s)鏡像之重導不追隨(視為拒用)。
 *
 * 併發: 同程序內同一最終檔且來源設定相同之併發呼叫共用同一次取得(其onProgress只通知最先呼叫者); 跨程序以暫存檔與原子更名保證不讀到半檔, 更名失敗而最終檔已是期望內容視為成功。各檔依清單順序逐一取得。
 *
 * 錯誤: reject帶code之Error: 'E_ARG'(參數)、'E_MANIFEST'(清單, 帶field)、'E_OFFLINE_MISSING'(離線缺檔, 帶missing)、'E_EXTRACT'(解壓)、'E_RENAME'(移入快取), 以及fsDownloadFileVerify之code('E_HTTP'、'E_NETWORK'、'E_TIMEOUT'、'E_TIMEOUT_TOTAL'、'E_SIZE_MISMATCH'、'E_HASH_MISMATCH'、'E_FS'等, 訊息前綴fsEnsureBins[<pkg>] file[<name>] from <github|url>); 皆帶step, 屬某檔者另帶pkg與bin(檔名; 檔案組為組名, 另帶file[組內直接來源檔之錯誤與解出成員之不符為組內路徑; 組之壓縮檔與分割檔之取得錯誤為該壓縮檔或分割檔之檔名; 解壓失敗只在該批只有一個成員時帶其組內路徑], 訊息前綴fsEnsureBins[<pkg>] bundle[<組名>] file[<同file之值>])。訊息與欄位內之網址帳密一律遮蔽(鏡像網址帶帳密者須傳入opt.fetch處理認證, 否則E_ARG)。
 *
 * 破壞性變更(wsemi-run10, 規則帳本sgud/rule/rulebook_breaking.md第42項): 網址之釘選判定收緊(只收版本標籤之資產網址與40碼提交之原始檔網址); 名稱leftovers、manifest、entries、bundles拒收; 主幹模式於主幹清單取不到時先用鏡像之manifest-last.json, 取得之檔不符時另連api.github.com重取; E_OFFLINE_MISSING之missing各項另有bundle; 結果另有不可列舉之manifest、entries、bundles。
 *
 * Unit Test: {@link https://github.com/yuda-lyu/wsemi/blob/master/test/fsEnsureBins.test.mjs Github}(檔案組: test/fsEnsureBinsBundle.test.mjs; 主幹模式、網址規則與採用之清單: test/fsEnsureBinsTrunk.test.mjs)
 * @memberOf wsemi
 * @param {Object|String} manifest 輸入清單物件(schema 1), 或其JSON檔路徑(相對路徑以opt.pkgRoot為基準)
 * @param {Object} opt 輸入設定物件
 * @param {String} opt.pkgRoot 輸入套件根目錄, 必填
 * @param {String|Array} [opt.names=null] 輸入只取部分檔之名稱或其陣列, 預設null為全部
 * @param {String} [opt.cacheDir=''] 輸入快取資料夾(其下為<sha256>/<name>), 優先於W_BIN_CACHE, ''為預設位置, 預設''
 * @param {String} [opt.mirror=undefined] 輸入鏡像(本地資料夾或http(s)基底網址), 給了即優先於W_BIN_MIRROR(空字串為不使用鏡像)
 * @param {Boolean} [opt.offline=undefined] 輸入是否離線, 給了即優先於W_BIN_OFFLINE
 * @param {Function} [opt.fetch=null] 輸入fetch函數(測試注入或處理認證), 未給時用全域fetch
 * @param {Function} [opt.onProgress=null] 輸入進度通知函數, 收到{ pkg, step('manifest'、'cache'、'mirror'、'download'、'extract'、'publish'[檔案組整組發布]、'ready'、'cleanup'), name, ... }, 其拋錯皆忽略
 * @param {Function} [opt.onTrace=null] 輸入觀察回調函數, 於內部事件發生當下同步呼叫, 傳入一個凍結之事件物件{ type, ... }, 供測試以事件(而非等待時間)同步與診斷; 與onProgress(給使用者之進度)分開, 不轉傳給內部之下載核心。type與其欄位為: 'manifest'(本次採用之清單已確定, 含source為'base'[釘選模式之傳入清單]、'remote'[取得主幹清單]、'mirror-last'[主幹模式改用鏡像之manifest-last.json]、'baseline'[主幹模式退回傳入之基準清單], 與files為該清單之項數[單一檔與檔案組]; 重取主幹清單後再通知一次), 'mirrorlast'(鏡像之manifest-last.json, 含result為'ok'、'miss'、'rejected'與code), 'refetch'(取得之檔不符而重取主幹清單, 含phase為'start'[code為不符之錯誤碼]、'ok'、'fail'[code為重取失敗之錯誤碼或null]), 'publish'(檔案組整組發布, 含name與result為'placed'、'replaced'[替換不符之舊組]、'exists'[他方已發布且整組校驗通過]), 檔案組之直接來源檔之'mirror'、'download'、'downloaded'之file為組內路徑, 其壓縮檔與分割檔者為該壓縮檔或分割檔之檔名(本地鏡像無組資料夾時'mirror'之file為null), 其'extract'、'extracted'為每批tar各一次, 'trunk'(主幹清單, 含phase為'fetch'[開始取, 時限計時器已建立]、'ok'[取得且檢核通過]、'fail'[失敗, 逾opt.stallMs而放棄者code為'E_TIMEOUT'], 與code為失敗之錯誤碼或null), 'cancel'(已取消主幹清單回應之本體, 含why為'unread'[讀取中途被中止]、'late'[放棄等待後才落定之回應]、'settled'[逾時勝出時已落定而未交出之回應]、'status'[非2xx之回應]), 'last'(主幹清單副本寫入manifest-last.json, 含ok), 'job'(各檔取得之登記, 含name與mode為'new'[新開一次取得]、'join'[共用同程序已在進行之同一次取得, 該次取得之其餘事件只通知最先呼叫者]、'solo'[選項無法序列化而不共用]), 'cache'(快取校驗結果, 含name與hit), 'mirror'(鏡像嘗試, 含name、file與result為'ok'、'miss'、'rejected'), 'offline'(離線缺檔之判定, 含name), 'work'(已建本次專用暫存夾, 含name), 'download'(開始自GitHub或廠商網址下載, 含name與file), 'downloaded'(下載結束, 含name、file、ok與code), 'extract'(開始解壓, 含name), 'kill'(解壓逾opt.totalMs而即將終止解壓程式, 含name), 'extracted'(解壓程式已結束或放棄等待, 含name、result為'exit'、'timeout'、'error', code為離開碼或null, exited為是否確認已結束或null), 'place'(解出檔已移入快取, 含name), 'cleanup'(暫存夾刪除結果, 含name與ok; 刪不掉或因解壓程式未結束而保留者ok為false), 'ready'(該檔已就緒, 含name), 'fail'(該檔失敗[離線缺檔除外], 含name與code), 'tidy'(離線缺檔reject前清除本次新建之空資料夾, 含n為清除個數), 'settle'(整體結果, 含ok與code; ok為false時code為錯誤碼或null)。事件皆於對應動作之後、resolve或reject之前通知, 回調不帶接收者呼叫, 其拋錯與回傳值皆忽略, 不影響取得, 預設null
 * @param {String} [opt.floatingManifestUrl=null] 輸入主幹模式之清單網址, 須在https://raw.githubusercontent.com/<repo>/之下, 預設由清單之repo與manifestPath組成
 * @param {Integer} [opt.maxAttempts=3] 輸入下載之最多嘗試次數(交fsDownloadFileVerify)
 * @param {Number} [opt.retryDelayMs=1000] 輸入下載之重試間隔基數毫秒(交fsDownloadFileVerify)
 * @param {Number} [opt.stallMs=30000] 輸入下載之停滯毫秒數(交fsDownloadFileVerify)
 * @param {Number} [opt.totalMs=600000] 輸入每個檔下載之整體時間上限毫秒數(交fsDownloadFileVerify)
 * @param {Object} opt.fs 輸入fs, 須有mkdirSync、lstatSync、openSync、closeSync、writeSync、readFileSync、createReadStream、renameSync、unlinkSync、rmSync及fsDownloadFileVerifyCore所需者
 * @param {Object} opt.path 輸入path, 須有resolve、join
 * @param {Object} opt.crypto 輸入crypto, 須有createHash
 * @param {Function} opt.Readable 輸入stream之Readable, 須有toWeb、fromWeb、from
 * @param {Function} opt.pipeline 輸入stream/promises之pipeline
 * @param {Function} opt.spawn 輸入child_process之spawn(解壓用)
 * @param {Object} [opt.env={}] 輸入環境變數物件(讀W_BIN_CACHE、W_BIN_MIRROR、W_BIN_OFFLINE、SystemRoot)
 * @returns {Promise} 回傳Promise, resolve回傳物件{ [name]: 絕對路徑 }(檔案組為其組資料夾), 另有不可列舉之屬性: leftovers(刪不掉之暫存夾[{ name, fp, error }])、manifest(本次採用之清單, 凍結: { source('base'、'remote'、'mirror-last'、'baseline'), url(主幹清單或鏡像之manifest-last.json之位置, 否則null), refetched(是否重取主幹清單), pkg, mode, repo, commit, manifestPath, version, note })、entries(所取各項之清單項{ [name]: 依清單之形式, sha256為小寫 }, 凍結, 供呼叫端知所用版本)、bundles(各檔案組{ [name]: { dir, id, files: { [組內路徑]: 絕對路徑 } } }, 凍結), reject回傳帶code之Error
 * @example
 * //need test in nodejs
 *
 * //see fsEnsureBins
 *
 */
async function fsEnsureBinsCore(manifest, opt = {}) {

    //觀察回調(opt.onTrace), 未給時為null; 取於buildCtx之前使E_ARG亦通知settle; opt不可讀(getter拋錯、已撤銷之Proxy)時視為未給, 其錯誤仍由buildCtx於讀取opt時照舊回報
    let trace = null
    try {
        trace = getTrace(opt)
    }
    catch (err) {}

    let ctx = null
    try {
        ctx = buildCtx(opt)
        ctx.trace = trace
        return await ensureMain(ctx, manifest, trace)
    }
    catch (err) {
        let e = ctx !== null ? withLeftovers(ctx, err) : err //任一失敗出口(含清單重取後之名稱選取): 附上本次各項刪不掉之暫存
        if (trace) {
            trace('settle', { ok: false, code: codeStr(e) })
        }
        throw e
    }
}


//ensureMain, fsEnsureBinsCore之主流程(成功時於回傳前通知settle, 失敗由呼叫端通知)
async function ensureMain(ctx, manifest, trace) {

    //清單, 物件或JSON檔路徑
    let raw = manifest
    let src = ''
    if (typeof manifest === 'string') {
        let fp = ctx.path.resolve(ctx.pkgRoot, manifest)
        src = ` ${fp}`
        try {
            raw = JSON.parse(String(ctx.fs.readFileSync(fp, 'utf8')).replace(/^\uFEFF/, ''))
        }
        catch (err) {
            throw mkErr('E_MANIFEST', `fsEnsureBins: can not read the manifest file "${fp}": ${errMsg(err)}`, { step: 'manifest', cause: err })
        }
    }
    let base = normManifest(raw, src, normOpt(ctx))
    ctx.pkg = base.pkg
    ctx.cacheRoot = ctx.cacheDir !== '' ? ctx.cacheDir : ctx.envCache !== '' ? ctx.path.join(ctx.envCache, base.pkg) : ctx.path.join(ctx.pkgRoot, 'bin-cache')

    //主幹模式取清單(主幹 → 鏡像之manifest-last.json → 基準清單)
    let res = await resolveManifest(ctx, base)
    let refetched = false
    let got = null
    try {
        got = await ensureAll(ctx, res, trace)
    }
    catch (err) {
        //主幹模式下取得之檔(或解出、組合之檔)與清單不符: 重取一次主幹清單(繞過CDN快取)再試, 仍不符才失敗(規格M5); 離線不重取
        //以本次採用之清單之模式判定(主幹清單或鏡像之manifest-last.json可為釘選模式之清單, 其不重取)
        let c = codeOf(err)
        if (base.mode !== 'floating' || res.m.mode !== 'floating' || ctx.offline || (c !== 'E_HASH_MISMATCH' && c !== 'E_SIZE_MISMATCH')) {
            throw err
        }
        if (trace) {
            trace('refetch', { phase: 'start', code: c })
        }
        //第一次嘗試之錯誤將被捨棄: 其刪不掉之暫存(下載核心之.part等)併入本次呼叫之leftovers
        try {
            for (let v of normLeft(ctx, Array.isArray(err.leftovers) ? err.leftovers : [], '')) {
                if (!ctx.leftovers.some((w) => keyOf(ctx, w.fp) === keyOf(ctx, v.fp))) {
                    ctx.leftovers.push(v)
                }
            }
        }
        catch (e2) {}
        let r2 = await refetchTrunk(ctx, base)
        if (r2 === null) {
            throw err
        }
        res = r2
        refetched = true
        got = await ensureAll(ctx, res, trace)
    }

    let r = Object.fromEntries(got.pairs)
    let mInfo = {
        source: res.source,
        url: res.url,
        refetched,
        pkg: res.m.pkg,
        mode: res.m.mode,
        repo: res.m.repo,
        commit: res.m.commit,
        manifestPath: res.m.manifestPath,
        version: res.m.version,
        note: res.m.note,
    }
    //鍵為呼叫端之名稱與路徑(可為__proto__), 一律以Object.fromEntries建立自有屬性
    let entries = Object.fromEntries(got.done.map(([e]) => {
        return [e.name, viewEntry(e)]
    }))
    let bundles = Object.fromEntries(got.done.filter(([e]) => e.kind === 'bundle').map(([e, fp]) => {
        let files = Object.fromEntries(e.files.map((f) => {
            return [f.path, ctx.path.join(fp, ...f.path.split('/'))]
        }))
        return [e.name, { dir: fp, id: e.id, files }]
    }))
    //不可列舉之屬性, 不影響{ [name]: 路徑 }之形狀(檔名與組名不得與之同名, 見normManifest): leftovers為刪不掉之暫存夾[{ name, fp, error }];
    //manifest為本次採用之清單(來源與頂層欄位), entries為所取各項之清單項(依清單之形式), bundles為各檔案組之{ dir, id, files: { [組內路徑]: 絕對路徑 } }
    Object.defineProperty(r, 'leftovers', { value: normLeft(ctx, ctx.leftovers, ''), enumerable: false }) //回傳時之狀態: 已不存在者不列
    Object.defineProperty(r, 'manifest', { value: deepFreeze(mInfo), enumerable: false })
    Object.defineProperty(r, 'entries', { value: deepFreeze(entries), enumerable: false })
    Object.defineProperty(r, 'bundles', { value: deepFreeze(bundles), enumerable: false })
    if (trace) {
        trace('settle', { ok: true, code: null })
    }
    return r
}


//ensureAll, 依採用之清單res(resolveManifest或refetchTrunk之結果)取得所選各項(opt.names, 預設全部), 回傳{ pairs: [[name, 路徑]], done: [[清單項, 路徑]] }
//各項依清單順序逐一取得; 離線缺檔者彙整為一個E_OFFLINE_MISSING
async function ensureAll(ctx, res, trace) {
    let m = res.m
    if (trace) {
        trace('manifest', { source: res.source, files: m.entries.length })
    }

    //names
    let byName = new Map(m.entries.map((e) => {
        return [e.name, e]
    }))
    let sel = m.entries
    if (ctx.names !== null) {
        let unknown = ctx.names.filter((n) => {
            return !byName.has(n)
        })
        if (unknown.length > 0) {
            throw mkErr('E_ARG', `fsEnsureBins[${ctx.pkg}]: opt.names has names not in the manifest: ${unknown.join(', ')} (the manifest has ${[...byName.keys()].join(', ')})`, { step: 'argument' })
        }
        sel = ctx.names.map((n) => {
            return byName.get(n)
        })
    }

    //逐一取得; 離線缺檔者彙整後一併回報
    let pairs = []
    let done = []
    let missing = []
    for (let e of sel) {
        let j = ensureEntry(ctx, e)
        try {
            let fp = await j.pm
            pairs.push([e.name, fp])
            done.push([e, fp])
        }
        catch (err) {
            if (isOwnErr(err) && err.code === 'E_OFFLINE_MISSING' && err.step === 'offline' && Array.isArray(err.missing) && err.pkg === undefined) {
                missing.push(...err.missing)
                continue
            }
            throw err
        }
        finally {
            ctx.leftovers.push(...j.left) //共用者共享同一次取得之leftovers
        }
    }
    if (missing.length > 0) {
        //離線缺檔: 不得留下本次建立之空資料夾(快取根、<sha256>夾、組資料夾), 先清掉其中仍為空者再回報
        let nTidy = tidyFresh(ctx)
        if (trace) {
            trace('tidy', { n: nTidy })
        }
        let err = offlineErr(ctx, missing)
        if (ctx.leftovers.length > 0) {
            err.leftovers = ctx.leftovers
        }
        throw err
    }
    return { pairs, done }
}


//deepFreeze, 深層凍結(回傳之唯讀資訊)
function deepFreeze(o) {
    for (let v of Object.values(o)) {
        if (v !== null && typeof v === 'object') {
            deepFreeze(v)
        }
    }
    return Object.freeze(o)
}


//viewSrc, 清單項之來源(依清單之形式): 自有檔為path, 廠商檔為url
function viewSrc(o) {
    return o.path !== null ? { path: o.path, sha256: o.sha256, size: o.size } : { url: o.url, sha256: o.sha256, size: o.size }
}


//viewArchive, 壓縮檔來源(單一壓縮檔或分割＋組合)之清單形式
function viewArchive(a) {
    if (a.archive !== null) {
        return viewSrc(a.archive)
    }
    return {
        parts: a.parts.map((p) => {
            return { name: p.name, ...viewSrc(p) }
        }),
        assembled: { type: 'zip', sha256: a.assembled.sha256, size: a.assembled.size },
    }
}


//viewEntry, 本次所取之清單項(依清單之形式, sha256為小寫, 分割檔之name為實際採用者), 供呼叫端記錄版本(如以sha256對照)
function viewEntry(e) {
    if (e.kind === 'bundle') {
        let v = { name: e.name }
        if (e.dir !== null) {
            v.dir = e.dir
        }
        let direct = e.files.filter((f) => {
            return f.src !== null
        })
        if (direct.length > 0) {
            v.files = direct.map((f) => {
                return f.src.path !== null ? { path: f.path, sha256: f.sha256, size: f.size } : { path: f.path, url: f.src.url, sha256: f.sha256, size: f.size }
            })
        }
        if (e.archives.length > 0) {
            v.archives = e.archives.map((a) => {
                let x = {
                    ...viewArchive(a),
                    extract: {
                        type: 'zip',
                        files: a.members.map((f) => {
                            return { path: f.path, sha256: f.sha256, size: f.size }
                        }),
                    },
                }
                if (a.note !== null) {
                    x.note = a.note
                }
                return x
            })
        }
        if (e.note !== null) {
            v.note = e.note
        }
        return v
    }
    let v = { name: e.name, ...(e.direct !== null ? viewSrc(e.direct) : viewArchive(e)) }
    if (e.extract !== null) {
        v.extract = { type: 'zip', member: e.extract.member, sha256: e.extract.sha256, size: e.extract.size }
    }
    if (e.note !== null) {
        v.note = e.note
    }
    return v
}
